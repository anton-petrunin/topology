\chapter{Coverings}\label{chap:coverings}

Now we will study covering maps --- a class of maps that mimic properties of the wrapping map $w\:\RR\to \SS^1$ from \ref{sec:lifting-s1}.
We used $w$ to calculate fundamental group of~$\SS^1$.
The covering maps will be used to calculate fundamental groups of other spaces.
But more importantly, these maps give another connection between algebra and geometry that has applications from both sides.

\section{Definitions}

\begin{thm}{Definition}\label{def:evenly-covered}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a continuous map.

\begin{wrapfigure}{r}{27 mm}
\vskip-2mm
\centering
\includegraphics{mppics/pic-1605}
\end{wrapfigure}

An open set $V\subset \c{Z}$ is called \index{evenly covered open set}\emph{evenly covered} by $w$ if
\[w^{-1}(V)=\bigcup_{\alpha\in \c{I}}\tilde V_\alpha,\]
where $\tilde V_\alpha$ are pairwise disjoint open sets in $\tilde{\c{Z}}$ such that restriction
$w|_{\tilde V_\alpha}$ is a homeomorphism $\tilde V_\alpha\to V$ for each $\alpha\in \c{I}$.
The sets $\tilde V_\alpha$ are called \index{slice}\emph{slices} over~$V$.

If for every point $z\in \c{Z}$ there is an open set $V\ni z$ that is evenly covered by $w$,
then $w$ is called a \index{covering map}\emph{covering} with \index{total space}\emph{total space}
$\tilde{\c{Z}}$ and \index{base space}\emph{base} $\c{Z}$.
\end{thm}

Note that the map $w\:\tilde{\c{Z}}\to \c{Z}$ is a covering, if $\c{Z}$ admits an open cover by evenly covered sets.%
\footnote{This phrase uses ``cover'' with two different meanings from \ref{def:evenly-covered} and \ref{def:cover}.}
Following the proof of \ref{prop:lifting}, one sees that the wrapping map $w(x)\z=\exp(2\cdot\pi\cdot i\cdot x)$  defines a covering map $\RR\to\SS^1$.
Keep this example in your head while reading further.

The following observation will be used without mentioning.

\begin{thm}{Observation}\label{obs:covering}
Let $V$ be an open set that is evenly covered by a continuous map $w$.
Then

\begin{subthm}{obs:covering:subset}
any open set $W\subset V$ is also evenly covered by $w$, and
\end{subthm}

\begin{subthm}{obs:covering:connected}
if $V$ is connected, then its slices are connected components of $w^{-1}(V)$.
\end{subthm}

\end{thm}




\begin{thm}{Observation}\label{thm:covering-open}
Any covering map is open.
\end{thm}

\parit{Proof.}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering map.
Choose an open set $\tilde V\subset \tilde{\c{Z}}$ and a point $\tilde z\in \tilde V$.

Choose an evenly covered open set $W\ni z=w(\tilde z)$.
The point $\tilde z$ must lie in one of its slices, say $\tilde W_\alpha$.
Let $\tilde W'_\alpha=\tilde W_\alpha\cap \tilde V$.
Since $w|_{\tilde W_\alpha}$ is a homeomorphism, $W'=w(\tilde W_\alpha')$ is open in $W$;
since $W$ is open,  we also have that $W'$ is open in $\c{Z}$.
That is, any $z\in w(\tilde V)$ admits an open neighborhood $W'\subset w(\tilde V)$;
hence $w(\tilde V)$ is open.
\qeds

\begin{thm}{Exercise}\label{ex:covering-clopen}
Let $w\:\tilde{\c{Z}}\to\c{Z}$ be a covering map.

\begin{subthm}{ex:covering-clopen:point}
Show that the inverse image $w^{-1}\{p\}$ of any one point set $\{p\}$ is a discrete subset of $\tilde{\c{Z}}$.
\end{subthm}


\begin{subthm}{ex:covering-clopen:image}
Show that the image $w(\tilde{\c{Z}})$ is a clopen set in $\c{Z}$.
Conclude that, if $\c{Z}$ is connected, and $\tilde{\c{Z}}\ne\emptyset$, then $w$ is surjective.
\end{subthm}

\end{thm}

Most interesting examples of coverings have connected total space $\tilde{\c{Z}}$ and base $\c{Z}$;
we will call them \emph{connected coverings}.
By \ref{ex:covering-clopen:image}, any connected covering is surjective.%
\footnote{Sometimes coverings are defined to be connected or surjective.}

\begin{thm}{Exercise}\label{ex:covering-product}
Let $v\:\tilde{\c{X}}\to \c{X}$ and $w\:\tilde{\c{Y}}\to \c{Y}$ be two covering maps.
Show that the map $\tilde{\c{X}}\times\tilde{\c{Y}} \to \c{X}\times\c{Y}$ defined by $(\tilde x,\tilde y)\mapsto (v(\tilde x),w(\tilde y))$ is a covering.
\end{thm}

\section{Trivial coverings}

Observe that any homeomorphism is a covering.
Further, suppose $\c{Z}$ is an arbitrary topological space and $\c{I}$ is a discrete space, then the projection $\c{Z}\times{\c{I}}\to \c{Z}$ defined by $(z,i)\mapsto z$ is a covering.
(Informally speaking, we took several disjoint copies of space $\c{Z}$ and map a point in each copy to the corresponding point in $\c{Z}$.)

A covering $w\:\tilde{\c{Z}}\to \c{Z}$ is called \index{trivial covering}\emph{trivial} if there is a discrete space $\c{I}$ and a homeomorphism $h\:\c{Z}\times{\c{I}}\to \tilde{\c{Z}}$ such that $w\circ h$ is the projection $\c{Z}\times{\c{I}}\to \c{Z}$.

{

\begin{wrapfigure}{r}{20 mm}
\vskip-0mm
\centering
\includegraphics{mppics/pic-1625}
\end{wrapfigure}

Note that any trivial connected covering must be a homeomorphism.
Since $\RR$ and $\SS^1$ are connected, the wrapping map $w(x)\z=\exp(2\cdot\pi\cdot i\cdot x)$ provides an example of a nontrivial covering map $\RR\to\SS^1$.
The map $z\mapsto z^n$ for an integer $n>1$ gives another example of nontrivial $n$-fold covering $\SS^1\to\SS^1$;
the total space and base are homeomorphic, but the map is not a homeomorphism.
The case $n=4$ is in the picture.

\section[$n$-fold coverings]{$\bm{n}$-fold coverings}

Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering.
If for some integer $n$ the inverse image of every point $z\in \c{Z}$ has exactly $n$ points, then we say that
$w$ is an \emph{$n$-fold} covering.

\begin{thm}{Exercise}\label{ex:n-folded}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a connected covering, and let $n$ be an integer.
Assume that an inverse image of some point $z\in \c{Z}$ has exactly $n$ points.
Show that $w$ is $n$-fold.
\end{thm}

\section{Fibered product}\label{sec:Fibered product}

Given two continuous maps $v\:\c{X}\to \c{Z}$ and $w\:\c{Y}\to \c{Z}$, consider the set
\[\c{P}=\set{(a,b)\in \c{X}\times \c{Y}}{v(a)=w(b)}\]
equipped with the induced topology from $\c{X}\times \c{Y}$.
Define $\tilde v\:\c{P}\z\to\c{Y}$ and $\tilde w\:\c{P}\to\c{X}$ as the restrictions to $\c{P}$ of
the projections $\c{X}\times \c{Y}\z\to \c{Y}$ and $\c{X}\times \c{Y}\to \c{X}$.

{

\begin{wrapfigure}[5]{l}{23 mm}
\vskip-3mm
\centering
\begin{tikzcd}
\c{P}\arrow[d, "\tilde w"'] \arrow[r, "\tilde v"] & \c{Y} \arrow[d, "w"]\\
\c{X} \arrow[r, "v"'] & \c{Z}
\end{tikzcd}
\end{wrapfigure}

The space $\c{P}$ equipped with the maps $\tilde v\:\c{P}\z\to\c{Y}$ and $\tilde w\:\c{P}\to\c{X}$ is called \index{fibered product}\emph{fibered product} of maps $v\:\c{X}\to \c{Z}$ and $w\:\c{Y}\to \c{Z}$.
The map $\tilde v$ is called \index{pullback}\emph{pullback} of $v$ along $w$;
similarly $\tilde w$ is a pullback of $w$ along $v$.
This is a natural choice of space and maps that makes the diagram commute. % \index{pullback}\emph{pullback diagram}.

}

{

\begin{wrapfigure}{r}{30 mm}
\vskip-4mm
\centering
\begin{tikzcd}
\c{X}\times \c{Y}\arrow[d] \arrow[r] & \c{Y} \arrow[d]\\
\c{X} \arrow[r] & *
\end{tikzcd}
\end{wrapfigure}

If $\c{Z}$ is a one-point space (denoted by $*$ on the diagram), then the maps $v$ and $w$ are uniquely defined.
In this case, $\c{P}=\c{X}\times \c{Y}$ and the maps $\tilde v$ and $\tilde w$ are projections, so fibered product generalizes the usual product.

}

\section{Lifts}

Earlier in \ref{sec:lifting-s1}, we defined lifts along the wrapping map $\RR\to\SS^1$.
Now we repeat the same definition for coverings.

\begin{wrapfigure}{r}{23 mm}
\vskip-0mm
 \centering
\begin{tikzcd}
& \tilde{\c{Z}} \arrow[d, "w"]\\
\c{Y} \arrow[r, "f"']\arrow[ru, "\tilde f"] & \c{Z}
\end{tikzcd}
\end{wrapfigure}

Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering, and let $f\:\c{Y}\to \c{Z}$ be a continuous map.
A continuous map $\tilde f\:\c{Y}\to \tilde{\c{Z}}$ is called a \index{lift}\emph{lift} of $f$ (along $w$) if $w\circ \tilde f=f$,
which means that the diagram commutes.

In particular, if $\c{Y}=[0,1]$, then $f$ is a path, and the path $\tilde f\:[0,1]\to \tilde{\c{Z}}$ is its \emph{lift}.
Furthermore, taking $\c{Y}=[0,1]\times [0,1]$, we can talk about lifts of homotopies of paths.

\begin{thm}{Lemma}\label{lem:path-lifting}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering map, and let $a_0\:[0,1]\z\to \c{Z}$ be a path.
Suppose that $\tilde z_0\in \tilde{\c{Z}}$ and $z_0\in \c{Z}$ be points such that $w(\tilde z_0)=z_0=a_0(0)$.
Then there is a unique lift $\tilde a_0$ of $a$ such that $\tilde a_0(0)=\tilde z_0$.

Moreover, for any homotopy of paths $a_\tau$ from $a_0$ to $a_1$ there is a unique lifted homotopy of paths $\tilde a_\tau$ such that $\tilde a_0(0)=\tilde z_0$ (and therefore $\tilde a_\tau(0)=\tilde z_0$ for any $\tau$).
In particular, if $a_0\sim a_1$, then $\tilde a_0(1)= \tilde a_1(1)$.
\end{thm}

This lemma is a direct generalization of \ref{prop:lifting}.
We omit the proof since it goes along the same lines.
In fact, the definition of covering map was made just to make this lemma hold.

\begin{thm}{Exercise}\label{ex:unique-lift}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering, and let $f\:\c{Y}\to \c{Z}$ be continuous maps.
Assume $\c{Y}$ is path-connected and $\tilde f_1,\tilde f_2\:\c{Y}\to \tilde{\c{Z}}$ are two lifts of $f$ along $w$.
Show that if $\tilde f_1(y_0)=\tilde f_2(y_0)$ for some $y_0\in \c{Y}$, then $\tilde f_1(y)=\tilde f_2(y)$ for any $y\in \c{Y}$.
\end{thm}


\begin{thm}{Theorem}\label{thm:bijection-fiber}
Let $w\: \tilde{\c{Z}}\to \c{Z}$ be a covering.
Assume that $\tilde{\c{Z}}$ is simply-connected.
Then, for any $z_0\in \c{Z}$ there is a bijection $\alpha\z\leftrightarrow \tilde z_\alpha$ between the fundamental group $\pi_1(\c{Z},z_0)$ and the inverse image $w^{-1}\{z_0\}\subset \tilde{\c{Z}}$.

Moreover, the bijection can be constructed in the following way:
Fix $\tilde z_1\in w^{-1}\{z_0\}$ and choose aloop $a\:[0,1]\z\to\c{Z}$ based at $z_0$;
let $\alpha=[a]\in \pi_1(\c{Z},z_0)$.
Consider a lift $\tilde a$ such $\tilde a(0)=\tilde z_1$ and let $\tilde z_\alpha= \tilde a(1)$.
\end{thm}

\parit{Proof.}
Note that $w(\tilde a(1))=a(1)=z_0$; therefore $\tilde z_\alpha\in w^{-1}\{z_0\}$ as required.

Let us show that $\alpha\mapsto\tilde z_\alpha$ is well-defined;
that is, the point $\tilde a(1)$ does not depend on choice of the loop $a$, assuming that $\alpha=[a]$.
Indeed, if $\alpha=[b]$ for some loop $b$, then $a\sim b$, and by the second statement in the lemma, we get $\tilde a(1)=\tilde b(1)$.

Since $\tilde{\c{Z}}$ is simply-connected, there is a path $\tilde a$ from $\tilde z_1$ to any point in $w^{-1}\{z_0\}$.
Note that $a=w\circ\tilde a$ is a loop in $\c{Z}$ with base point $z_0$, and $\tilde a$ is its lift as in the construction.
This implies that the map $\alpha\mapsto\tilde z_\alpha$ is onto $w^{-1}\{z_0\}$.

It remains to check that the map $\alpha\mapsto \tilde z_\alpha$ is injective.
Suppose $\tilde a(1)=\tilde b(1)$ for two paths starting at $\tilde z_1$.
Since $\tilde{\c{Z}}$ is simply-connected,
we have that $\tilde a\sim \tilde b$; that is, there is a homotopy $\tilde a_\tau$ from $\tilde a$ to~$\tilde b$.
Observe that $a_\tau=w\circ  \tilde a_\tau$ is a homotopy from $a=w\circ \tilde a$ to $b=w\circ \tilde b$;
that is, $a\sim b$ which is equivalent to $[a]=[b]\z\in \pi_1(\c{Z},z_0)$.
\qeds


\begin{thm}{Exercise}\label{ex:induced-hom-covering}
Let $w\:\tilde{\c{Z}}\to \c{Z}$ be a covering map.
Suppose that $\tilde z_0\in \tilde{\c{Z}}$ and $z_0\in \c{Z}$ be a points such that $w(\tilde z_0)=z_0$.

\begin{subthm}{ex:induced-hom-covering:injective}
Show that $w_*\:\pi_1(\tilde{\c{Z}},\tilde z_0)\to \pi_1(\c{Z}, z_0)$ is a \index{monomorphism}\emph{monomorphism} (which means injective homomorphism).
\end{subthm}

\begin{subthm}{ex:induced-hom-covering:lift}
Let $b$ be a loop in $\c{Z}$ with base point $z_0$ and $\tilde b$ be its lift in $\tilde{\c{Z}}$ such that $\tilde b(0)=\tilde z_0$.
Show that $\tilde b$ is a loop if and only if $[b]\in \Im w_*$.
\end{subthm}

\begin{subthm}{ex:induced-hom-covering:action}
Show that the following construction defines a right action of $\pi_1(\c{Z},z_0)$ on $w^{-1}\{z_0\}$.%
\footnote{This is the so-called \index{monodromy action}\emph{monodromy action}.}
Given $\alpha=[a]\in \pi_1(\c{Z},z_0)$ and a point $p\in w^{-1}\{z_0\}$, consider a lift $\tilde a$ of $a$ such that $\tilde a(0)=p$ and define $p\cdot\alpha \df\tilde a(1)$.
\end{subthm}

\end{thm}

\chapter{Normal coverings}

A normal covering is a covering that can be realized as a quotient map by a group action.
This class of coverings is particularly important in many applications.

\section{Freely discontinuous action}

A continuous action $G\acts \c{X}$ is called \index{freely discontinuous action}\emph{freely discontinuous} if for any $x\in\c{X}$ there is an open neighborhood $V \ni x$ such that
$V\cap g\cdot V=\emptyset$ for any $g\ne 1$.
In this case, we also have
\[
g_1\cdot V\cap g_2\cdot V=g_1\cdot (V\cap g_1^{-1}\cdot g_2\cdot V)=\emptyset
\]
for any tow distinct elements $g_1,g_2\in G$.

\begin{thm}{Proposition}\label{prop:freely-discontinuous}
Let $G\acts \c{X}$ be a continuous action.
The quotient map $w\:\c{X}\to \c{X}/G$ is a covering if and only if the action is freely discontinuous.
\end{thm}

If a covering $w$ can be realized as a quotient map by an action of some group $G$, then we say that $w$ is a \index{normal covering}\emph{normal covering}%
\footnote{If $w\:\tilde{\c{Z}}\to \c{Z}$ is a normal covering and $w(\tilde z_0)=z_0$, then $w_*(\pi_1(\tilde{\c{Z}},\tilde z_0))$ is a normal subgroup of $\pi_1(\c{Z}, z_0)$ --- this is the reason for the name.
Normal coverings are also called \index{regular covering}\emph{regular} and \index{Galois covering}\emph{Galois}.} or \index{$G$-covering}\emph{$G$-covering};
the latter term is used if one needs to specify the group.

Note that the map $w\:\RR\to \SS^1$ constructed in \ref{sec:lifting-s1} is a normal $\ZZ$-covering, where $\ZZ$ denotes the set of integers;
it is the quotient map for a freely discontinuous action $\ZZ\acts\RR$ by integer shifts $x\mapsto x+n$, where $n\in\ZZ$.

\parit{Proof.}
By the definition of the quotient map, $w\:\c{X}\to \c{X}/G$ is continuous and surjective;
moreover, by \ref{ex:quotient-map:open}, $w$ is open.
Let $V$ be an open neighborhood of $x\in \c{X}$ that meets the condition in the definition of freely discontinuous action.

Since $w$ is open, $W=w(V)$ is an open neighborhood of $w(x)\z=[x]=G\cdot x$.
Furthermore,
\[w^{-1}(W)=G\cdot V=\bigcup_{g\in G} g\cdot V.\]
and the restriction $w|_{g\cdot V}$ is a homeomorphism $g\cdot V\to W$;
indeed, $w|_{g\cdot V}$ is open, continuous, and bijective map $g\cdot V\to W$.
Therefore, $w$ meets the conditions in the definition of covering.

On the other hand, if the quotient map $w\:\c{X}\to \c{X}/G$ is a covering,
then every point $w(x)\in \c{X}/G$ admits an evenly covered neighborhood $W$;
its slice $V\ni x$ meets the condition in the definition of freely discontinuous action.
\qeds

\section{Finite group}

An action $G\acts \c{X}$ is called \index{free action}\emph{free} if $g\cdot x\ne x$ for any $x\in\c{X}$ and $g\ne 1$.
Note that any freely discontinuous action is free, but not the other way around;
for example, the action $\RR\acts\RR$ by shifts $x\mapsto x+c$ is free, but not freely discontinuous.

\begin{thm}{Corollary}\label{cor:covering-finite-action}
Let $G\acts \c{X}$ be a continuous free action of a finite group $G$.
Assume that $\c{X}$ is Hausdorff.
Then the quotient map $\c{X}\z\to \c{X}/G$ is a covering.
\end{thm}

\parit{Proof.}
Choose $x\in \c{X}$ and $g\ne 1$ in $G$.
Since $G\acts \c{X}$ is free, $g\cdot x\ne x$.
Since $\c{X}$ is Hausdorff we can choose disjoint open sets $U_g\ni x$ and $V_g\z\ni g\cdot x$.
Let $W_g=U_g\cap g^{-1}\cdot V_g$.
Note that $W_g\ni x$ and $g\cdot W_g\ni g\cdot x$ are disjoint open sets.
Indeed, $W_g\cap g\cdot W_g\subset U_g\cap V_g=\emptyset$.

Now consider the finite intersection of open sets
\[W=\bigcap_{g\ne 1}W_g\ni x,\]
which is open.
Note that $g\cdot W$ are disjoint for all $g\in G$;
that is, the action $G\acts \c{X}$ is freely discontinuous.
It remains to apply \ref{prop:freely-discontinuous}.
\qeds

\section{Real projective plane}

The equivalence class of an integer $m$ modulo $n$ will be denoted by $[m]_n$;
that is,
\[[m]_n =\set{m+k\cdot n}{k\in\ZZ};\]
here $\ZZ$ denotes the set of integers.
The set of equivalence classes of integers modulo $n$ will be denoted by $\ZZ_n$.
The operation
\[[m_1]_n\z+[m_2]_n\df[m_1+m_2]_n\]
defines on $\ZZ_n$ a group structure with neutral element $[0]_n$;
it is \index{cyclic group}\emph{cyclic group} of order $n$, generated by element $[1]_n$.

Recall that all cyclic groups of order $n$ are isomorphic, so it is acceptable to denote a cyclic groups of order $n$ by $\ZZ_n$.

Consider the action $\ZZ_2\acts\SS^2$ generated by the involution $x\mapsto -x$;
note that this action is free.
The quotient space $\SS^2/\ZZ_2$ is called \emph{real projective plane} and denoted by $\RP^2$.
Since $\ZZ_2$ is finite, by \ref{cor:covering-finite-action}, the quotient map $\SS^2\to \RP^2$ is a covering.
By \ref{ex:van-kamp}, $\SS^2$ is simply-connected.
By \ref{thm:bijection-fiber}, $\pi_1(\RP^2)$ has exactly two elements, and therefore it is isomorphic to $\ZZ_2$.
(This is the only group with two elements up to isomorphism.)
One could also deduce that $\pi_1(\RP^2)\cong\ZZ_2$ from \ref{ex:monodromy:isomorphism}.
One way or another, we proved the following.

\begin{thm}{Claim}
$\pi_1(\RP^2)\cong \ZZ_2$.
\end{thm}

\section{Borsuk--Ulam theorem}

\begin{thm}{Theorem}\label{thm:Borsuk-Ulam}
There is no continuous map $\tilde h\:\SS^2\to \SS^1$ such that $\tilde h(-\tilde x)=-\tilde h(\tilde x)$ for any $\tilde x\in\SS^2$.
\end{thm}

The proof relies on the following exercise and lemma.

\begin{thm}{Exercise}\label{ex:2-folded-s1}
Let $\ZZ_2\acts \SS^1$ be the action generated by involution $x\mapsto -x$.
Show that $\SS^1/\ZZ_2\simeq\SS^1$ and therefore the quotient map defines a 2-fold cover $\SS^1\to \SS^1$.
\end{thm}

\begin{thm}{Lemma}
Any homomorphism $\ZZ_2\to\ZZ$ is \index{trivial homomorphism}\emph{trivial}.
\end{thm}

Note that the homomorphism in the lemma is trivial if and only if $[1]_2\mapsto 0$.

\parit{Proof.}
Assume $\phi\:\ZZ_2\to\ZZ$ is a nontrivial homomorphism,
then $\phi[1]_2\z=n$ for an integer $n\ne 0$.
But then
\[
0=\phi([0]_2)=\phi([1]_2+[1]_2)=\phi([1]_2)+\phi([1]_2) =n+n\ne 0
\]
--- a contradiction.
\qeds


Most of the proof of \ref{thm:Borsuk-Ulam} will be defining and explaining the following commutative diagram.

\parit{Proof of \ref{thm:Borsuk-Ulam}.}
Assume that $\tilde h$ exists.


Choose $\tilde x_0\in \SS^2$ and a path $\tilde a\:[0,1]\to \SS^2$ from $\tilde x_0$ to $-\tilde x_0$.
Let $v\: \SS^2\to\RP^2$ be the quotient map for the equivalence relation $\tilde x\sim -\tilde x$.
Recall that $v$ is a 2-fold covering.
Note that $a=v\circ \tilde a$ is a loop in $\RP^2$ based at $x_0=v(\tilde x_0)$, and $\tilde a$ is its lift.%
\footnote{By \ref{thm:bijection-fiber}, we have $\alpha=[a]=[1]_2\in \ZZ_2=\pi_1(\RP^2)$, but it will not be used in the proof directly.}

The path $\tilde b=\tilde h\circ \tilde a$ runs from $\tilde y_0=\tilde h(\tilde x_0)$ to $-\tilde y_0=-\tilde h(\tilde x_0)$.
Let $w\: \SS^1\to\SS^1$ be the quotient map for the equivalence relation $\tilde y\sim -\tilde y$, which is also a 2-fold covering map (see \ref{ex:2-folded-s1});
note that $b=w\circ \tilde b$ is a loop in $\SS^1$ based at $y_0=w(\tilde y_0)$ and $\tilde b$ is its lift to $\SS^1$.
Since $\tilde b(1)\ne \tilde b(0)$, \ref{thm:bijection-fiber} implies that $\beta=[b]\ne 0\in \ZZ=\pi_1(\SS^1)$.

Recall that $v\: \SS^2\to\RP^2$ and $w\: \SS^1\to\SS^1$ are quotient maps that identify antipodal points.
Since $\tilde h$ is odd, it maps the equivalence class in $\SS^2$ to an equivalence class in $\SS^1$.
In other words, there is a map $h\:\RP^2\to\SS^1$ such that $h\circ v=w\circ \tilde h$.%
\footnote{The dashed arrow on the diagram encodes the claim that $h$ exists.}

We have finished describing the diagram and now the proof starts.
Consider the induced homomorphism
\[h_*\:\pi_1(\RP^2,x_0)\to \pi_1(\SS^1,y_0).\]
Since $\pi_1(\RP^2,x_0)\cong \ZZ_2$ and $\pi_1(\SS^1,y_0)\cong \ZZ$, the lemma implies that $h_*$ is trivial.
On the other hand $h_*[a]=[b]$.
From above, $[b]\ne 0\in\ZZ$ --- a contradiction.
\qeds

\begin{subthm}{ex:Borsuk-Ulam:Borsuk-Ulam}
Show that for every continuous map $f\: \SS^2\to \RR^2$ there exists a pair of antipodal
points in $\SS^2$ that map to a single point.
\end{subthm}

\end{subthm}

to conclude that one of the sets $A$, $B$, or $C$
contains a pair of antipodal
points in $\SS^2$.

\end{subthm}

\end{thm}


\section{Monodromy}\label{sec:monodromy}

\begin{thm}{Proposition}
Let $w\:\tilde{\c{Z}}\to\c{Z}$ be a normal $G$-covering.
Suppose that $w(\tilde z_1)=z_0$ for some points $\tilde z_1\in \tilde{\c{Z}}$ and $z_0\in \c{Z}$.
Then there is a unique homomorphism $m\:\pi_1(\c{Z},z_0)\to G$ such that
$m[a]\cdot \tilde z_1=\tilde a(1)$ for any loop $a\:[0,1]\to \c{Z}$ based at $z_0$ and its lift $\tilde a\:[0,1]\to \tilde{\c{Z}}$ such that $\tilde a(0)=\tilde z_1$.

\end{thm}

The homomorphism $m\:\pi_1(\c{Z},z_0)\to G$ is called \index{monodromy homomorphism}\emph{monodromy homomorphism} of the normal $G$-covering $w\:\tilde{\c{Z}}\to\c{Z}$.

\parit{Proof.}
Note that $w^{-1}\{z_0\}=G\cdot\tilde z_1$.

Choose $\alpha=[a]\in \pi_1(\c{Z},z_0)$, let $\tilde a$ be  a liftof $a$ such $\tilde a(0)=\tilde z_1$.
Note that $w\circ \tilde a(1)=a(1)=z_0$, so $\tilde a(1)=g_\alpha\cdot \tilde z_1$ for some $g_\alpha\in G$.

By \ref{lem:path-lifting}, $\tilde a(1)$ and therefore $g_\alpha$ depends only on the homotopy class $\alpha=[a]$.
Thus we get a well-defined map $\alpha\mapsto g_\alpha$ from $\pi_1(\c{Z},z_0)$ to $G$.

Let us show that this map is a homomorphism $\pi_1(\c{Z},z_0)\to G$.
In other words, we need to show that
\[g_{\alpha\cdot\beta}\z=g_\alpha\cdot g_\beta.\]

Choose loops $a$ and $b$ based at $z_0\in\c{Z}$ such that $\alpha=[a]$ and $\beta=[b]$.
Let $\tilde a$ and $\tilde b$ be the lifts of $a$ and $b$ that start at $\tilde z_1$.
Note that $g_\alpha\circ \tilde b$ is a lift of $b$ that starts at $\tilde a(1)=g_\alpha\cdot \tilde z_1$.
Therefore $\tilde a*(g_\alpha\circ \tilde b)$ is a lift of $a*b$ that start at $\tilde z_1$.
It follows that
\[g_{\alpha\cdot \beta}(\tilde z_1)=\tilde a*(g_\alpha\circ \tilde b)(1)=g_\alpha\circ\tilde b(1)=g_\alpha\circ g_\beta(\tilde z_1).\]
It remains to set $m(\alpha)\df g_\alpha$.
\qeds

\begin{thm}{Exercise}\label{ex:monodromy}
Let $m\:\pi_1(\c{Z},z_0)\to G$ be a monodromy homomorphism of a normal $G$-covering $w\:\tilde{\c{Z}}\to\c{Z}$.
\begin{subthm}{ex:monodromy:surjective}
Assume $\tilde{\c{Z}}$ is path-connected.
Show that $m$ is surjective.
\end{subthm}

\begin{subthm}{ex:monodromy:isomorphism}
Assume $\tilde{\c{Z}}$ is simply-connected.
Show that $m$ is an isomorphism.
\end{subthm}

\begin{subthm}{ex:monodromy:trivial}
Assume $w$ is a trivial covering. Show that $m$ is \index{trivial homomorphism}\emph{trivial}; that is it sends the whole fundamental group to the nutral element of $G$.
\end{subthm}

\end{thm}

\section{Klein bottle}

Consider the following tesselation of $\RR^2$ into unit squares.
\begin{figure}[!ht]
\vskip -0mm

\includegraphics{mppics/pic-1740}
\vskip -6mm

Let $G\acts\RR^2$ be the group action that preserves this tesselation and the arrows on the sides of squares.
Formally speaking, the action includes the motions
\[(x,y)\mapsto (m+x, n+(-1)^m\cdot y)\]
for all integers $m$ and $n$.

The quotient space $\RR^2/G$ is called \emph{Klein bottle}.
\begin{figure}[h!]

\includegraphics{mppics/pic-1743}
\hskip 4mm
\includegraphics[angle=90]{asy/klein-bottle}

This space can be also glued from the square $[-1,1]\times[-1,1]$ by the minimal equivalence relation such that $(x,-1)\sim (x,1)$ and $(-1,y)\sim (1,-y)$.
The picture on the right might help to visualize this space, but this is not an embedding of $\c{K}$, and in fact there is embedding of $\c{K}$ in $\RR^3$.

\begin{thm}{Exercise}
Let $G\acts \RR^2$ be the described action,
and let $\c{K}\z=\RR^2/G$ be the Klein bottle.

Show that the action $G\acts \RR^2$ is freely discontinuous.
\end{subthm}

\end{subthm}

\end{subthm}

\end{subthm}

\end{thm}
