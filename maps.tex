\section{Homeomorphisms}

A bijection $f\:\c{X}\to\c{Y}$ between topological spaces 
is called a \index{homeomorphism}\emph{homeomorphism} if $f$ and its inverse $f^{-1}\:\c{Y}\to\c{X}$
are continuous.%
\footnote{The terms \textit{homomorphism} and \textit{homeomorphism} look similar, and they have
similar, but different meanings. Homomorphisms are creatures from abstract algebra; we will need them starting from Chapter~\ref{chap:algebra}.}

Topological spaces $\c{X}$ and $\c{Y}$ are called \index{homeomorphic spaces}\emph{homeomorphic} (briefly, $\c{X}\simeq\c{Y}$) if there is a homeomorphism $f\:\c{X}\to\c{Y}$.

A map $f\:\c{X}\to \c{Y}$ is called an \index{embedding}\emph{embedding} if $f$ defines a homeomorphism from $\c{X}$ to the subspace $f(\c{X})$ in $\c{Y}$.

\begin{thm}{Exercise}\label{ex:bijection-ne-homeo}
Show that any homeomorphism is a continuous bijection.

Give an example of a continuous bijection between topological spaces that is not a homeomorphism.
\end{thm}

\begin{thm}{Exercise}\label{ex:exp}
Show that $x\mapsto e^x$ is a homeomorphism $\RR\to (0,\infty)$.
\end{thm}

\begin{thm}{Exercise}\label{ex:arctan}
Construct a homeomorphism $f\:\RR\to (0,1)$.
\end{thm}

\begin{thm}{Exercise}\label{ex:homeo=eq}
Show that $\simeq$ is an equivalence relation;
that is, for any topological spaces $\c{X}$, $\c{Y}$, and $\c{Z}$ we have the following:
\begin{subthm}{}
$\c{X}\simeq\c{X}$;
\end{subthm}

\begin{subthm}{}
if $\c{X}\simeq\c{Y}$, then $\c{Y}\simeq\c{X}$;
\end{subthm}

\begin{subthm}{}
if $\c{X}\simeq\c{Y}$ and $\c{Y}\simeq\c{Z}$, then $\c{X}\simeq\c{Z}$.
\end{subthm}

\end{thm}

\begin{thm}{Advanced exercise}\label{ex:square}
Let $Q\subset \RR^2$ be a solid square with side $S$ and vertex $v$.
Show that $Q\setminus S \simeq Q\setminus \{v\}$.
\end{thm}


\begin{thm}{Advanced exercise}\label{ex:inversion}
Prove that the complement of a circle in $\RR^3$ is homeomorphic to $\RR^3$ without a line $\ell$ and a point $p\not\in\ell$.
\end{thm}

Recall that a set $V\subset\RR^2$ is called \index{star-shaped set}\emph{star-shaped} if there exists a point $p\in V$ such that for all $x\in V$ the line segment from $p$ to $x$ lies in~$V$.

\begin{thm}{Advanced exercise}\label{ex:star-shaped}
Show that any nonempty open star-shaped set in the plane is homeomorphic to the open disc.
\end{thm}

\begin{thm}{Advanced exercise}\label{ex:cont-dense}

\end{thm}

\section{Closed and open maps}
\label{sec:Closed and open maps}

\begin{thm}{Definition}\label{def:open-closed-maps}
Let $f\:\c{X}\to\c{Y}$ be a map between topological spaces;
$f$ is called \index{open map}\emph{open} if, for any open set $V\subset\c{X}$, the image $f(V)$ is open in~$\c{Y}$.
Similarly,
$f$ is called \index{closed map}\emph{closed} if, for any closed set $Q\subset\c{X}$, the image $f(Q)$ is closed in $\c{Y}$.
\end{thm}

Note that a bijective map is open if its inverse is continuous.
By \ref{prop:cont-closed}, we also have that a bijective map is closed if its inverse is continuous.
Thus we get the following.

\begin{thm}{Observation}\label{ex:bijective-closed-open}
A homeomorphism can be defined as a continuous open bijection as well as a continuous closed bijection.
\end{thm}


\begin{thm}{Exercise}\label{ex:closed-open-cont}
Give an example of a map $f$ between topological spaces such that
\begin{subthm}{}
$f$ is continuous and open, but not closed,
\end{subthm}

\begin{subthm}{}
$f$ is continuous and closed, but not open,
\end{subthm}

\begin{subthm}{}
$f$ is closed and open, but not continuous.
\end{subthm}

\end{thm}

\begin{thm}{Advanced exercise}\label{ex:closed-open-cont-R}
Construct a map $\RR\to\RR$ that is open, but not continuous.
\end{thm}
