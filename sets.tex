\chapter{Subsets}

This chapter starts with the definition of closed sets in a topological space.
Further, we introduce constructions of interior, closure, and boundary.
Finally, we define neighborhoods and discuss limits in a general topological space.

There is no particular reason why we define a topological space in terms of open sets --- we could use closed sets instead.
(In fact, closed sets were considered before open sets --- the former were introduced by Georg Cantor in 1884 \cite{cantor}, and the latter by René Baire in 1899 \cite{baire}.)

\section{Closed sets}\label{sec:closed-sets}

Let $\c{X}$ be a topological space. 
A subset $K\subset \c{X}$ is called \index{closed set}\emph{closed} if its complement $\c{X}\setminus K$ is open.

Sometimes it is easier to use closed sets; 
for example, the cofinite topology can be defined by declaring that the whole space and all its finite sets are closed.

From the definition of topological spaces, the following properties of closed sets follow.

\begin{thm}{Proposition}\label{prop:closed-properties}
Let $\c{X}$ be a topological space. 
\begin{subthm}{}
 The empty set and $\c{X}$ are closed.
\end{subthm}

\begin{subthm}{def:closed-properties:n}
The intersection of any family of closed sets is a closed set.
That is, if $K_\alpha$ is closed for any $\alpha$ in the index set $\c{I}$,
then the set
\[Q=\bigcap_{\alpha\in \c{I}} K_\alpha
=
\set{x\in \c{X}}{x\in K_\alpha\quad\text{for any}\quad\alpha\in \c{I}}\]
is closed.
\end{subthm}

\begin{subthm}{def:closed-properties:u}
The union of two closed sets (or any finite family of closed sets) is closed.
That is,  if $K_1$ and $K_2$ are closed, then the union $Q=K_1 \cup K_2$ is closed. 
\end{subthm}

\end{thm}

The following proposition is completely analogous to the original definition of continuous maps via open sets (\ref{def:cont-top}). 

\begin{thm}{Proposition}\label{prop:cont-closed}
Let $\c{X}$ and $\c{Y}$ be topological spaces.
A map $f\:\c{X}\z\to\c{Y}$ is continuous if and only if any closed set $Q\subset \c{Y}$ has a closed inverse image  $f^{-1}(Q)\subset\c{X}$.
\end{thm}


\parit{Proof.}
In the proof, we will use the following set-theoretical identity.
Suppose $A\subset \c{Y}$ and $B=\c{Y}\setminus A$ (or, equivalently, $A=\c{Y}\setminus B$).
Then
$$f^{-1}(B)
=
\c{X}\setminus f^{-1}(A)\eqlbl{eq:inverse-image-complement}$$
for any map $f\:\c{X}\to\c{Y}$.
This identity is tautological;
to prove it, observe that both sides can be spelled as
\[\set{x\in\c{X}}{f(x)\notin A}.\]

\parit{Only-if part.}
Let $B\subset \c{Y}$ be a closed set.
Then $A=\c{Y}\setminus B$ is open.
Since $f$ is continuous, 
$f^{-1}(A)$ is open.
By \ref{eq:inverse-image-complement},
$f^{-1}(B)$ is the complement of $f^{-1}(A)$ in $\c{X}$.
Hence $f^{-1}(B)$ is closed.

The only-if part follows since $B$ is an arbitrary closed set in $\c{Y}$.

\parit{If part.}
Fix an open set $B$; its complement $A=\c{Y}\setminus B$ is closed.
Therefore $f^{-1}(A)$ is closed.
By \ref{eq:inverse-image-complement},
$f^{-1}(B)$ is the complement of $f^{-1}(A)$  in~$\c{X}$.
Hence $f^{-1}(B)$ is open.

The if part follows since $B$ is an arbitrary open set in $\c{Y}$.
\qeds

\begin{thm}{Exercise}\label{ex:open-intersection+metrizable}

\begin{subthm}{ex:open-intersection+metrizable:a}
Let $\c{X}$ be a metrizable topological space.
Show that any closed set in $\c{X}$ is an intersection of a (countable) family of open sets.
\end{subthm}

\begin{subthm}{ex:open-intersection+metrizable:b}
Construct a topological space $\c{Y}$ with a closed set $Q$ that is not an intersection of any family of open sets.
\end{subthm}

\end{thm}


\section{Interior and closure}

Let $A$ be an arbitrary subset of a topological space $\c{X}$.

The union of all open subsets of $A$ is called the \index{interior}\emph{interior} of $A$ and denoted as $\mathring A$, for longer expressions we may use $A^\circ$ or $\interior A$.

Note that $\mathring A$ is open.
Indeed, it is defined as a union of open sets and such union is open by the definition of a topology (\ref{def:top-space}).
So we can say that $\mathring A$ is the \textit{maximal} open set in $A$, 
as any open subset of $A$ lies in~$\mathring A$.
In particular, $\mathring{\mathring A}=\mathring A$ for any set $A$.

The intersection of all closed subsets containing $A$ is called the \index{closure}\emph{closure} of $A$ and denoted as $\bar A$ or $\closure A$.

The set $\bar A$ is closed.
Indeed, it is defined as an intersection of closed sets and such intersection has to be closed by \ref{prop:closed-properties}.
In other words, $\bar A$ is the \textit{minimal} closed set that contains $A$, 
as any closed subset containing $A$ contains $\bar A$.
In particular, $\bar{\bar A}=\bar A$ for any set $A$.

\begin{thm}{Exercise}\label{ex:closure-interior-complement}
Assume $A$ is a subset of a topological space $\c{X}$;
consider its complement $B=\c{X}\setminus A$.
Show that 
\[\bar B=\c{X}\setminus \mathring A.\]
\end{thm}

%\begin{thm}{Exercise}\label{ex:closure-interior}Let $A$ and $B$ be subsets of a topological space $\c{X}$. Show that (a) $\overline A\cup B=\bar A\cup \bar B$, (b) $\overline A\cap B\subset \bar A\cap \bar B$, (c) $(A\cup B)^\circ\supset\mathring A\cup \mathring B$, (d) $(A\cap B)^\circ=\mathring A\cap \mathring B$, (e) show that the inclusions in (b) and (c) might be strict. \end{thm}

\section{Boundary}

Let $A$ be an arbitrary subset of a topological space $\c{X}$.
The \index{boundary}\emph{boundary}\footnote{Also known as \index{frontier}\emph{frontier}.} of $A$ (briefly $\partial A$) is defined as the complement
\[\partial A=\bar A\setminus \mathring A.\]

\begin{thm}{Exercise}\label{ex:bry-closed}
Show that the boundary of any set is closed.
\end{thm}

\begin{thm}{Exercise}\label{ex:bry<closure}
Show that the set $A$ is closed if and only if $\partial A\subset A$.
\end{thm}

\begin{thm}{Exercise}\label{ex:bry-bry}
Show that for any set $A$ in a topological space we have $\partial(\partial A)\subset \partial A$.
Provide an example showing that the inclusion might be strict.
\end{thm}

\begin{thm}{Advanced exercise}\label{ex:3=bry}
Find three disjoint open sets on the real line 
that have the same nonempty boundary.
\end{thm}

\section{Subspaces}\label{sec:induced-topology}

\begin{thm}{Proposition}\label{prop:subspace}
Let $A$ be a subset of a topological space $\c{Y}$.
Then all subsets $V\subset A$ such that $V=A\cap W$ for some open set $W$ in $\c{Y}$ form a topology on $A$.
\end{thm}

The described topology is called the \index{induced topology}\emph{induced topology} on $A$.

A subset $A$ of a topological space $\c{Y}$ equipped with the induced topology is called a \index{subspace}\emph{subspace} of $\c{Y}$.
It is straightforward to check that this notion agrees with the notion introduced in \ref{sec:subspaces-metric};
that is, if $\c{Y}$ is a metric space, then any subset $A\subset\c{Y}$ comes with a metric,
and the topology defined by this metric coincides with the induced topology on $A$.

\parit{Proof.}
We need to check the conditions in \ref{def:top-space}.

First, the whole set $A$ and the empty set are included;
indeed, $\emptyset=A\cap \emptyset$ and $A=A\cap \c{Y}$.

Assume $\{V_\alpha\}$ is a family of open sets in $A$; here $\alpha$ runs in some index set, say $\c{I}$.
In other words, for each $V_\alpha$ there is an open set $W_\alpha\subset\c{Y}$ such that $V_\alpha\z=A\cap W_\alpha$.
Note that
\[\bigcup_\alpha V_\alpha=A\cap\left(\bigcup_\alpha W_\alpha\right).\]
Since the union of $\{W_\alpha\}$ is open in $\c{Y}$ (\ref{def:top-space:u}), the union of $\{V_\alpha\}$ is open in the induced topology on $A$.

Assume $V_1$ and $V_2$ are open in $A$;
that is, $V_1=A\cap W_1$ and  $V_2\z=A\cap W_2$ for some open sets $W_1,W_2\subset\c{Y}$.
Note that
\[V_1\cap V_2=A\cap(W_1\cap W_2).\]
Since the intersection $W_1\cap W_2$ is open in $\c{Y}$ (\ref{def:top-space:n}),
the intersection $V_1\cap V_2$ is open in the induced topology on $A$.
\qeds

\begin{thm}{Exercise}\label{ex:rel-int/cl}
Let $\c{A}$ be a subspace in $\c{X}$.
Given a set $S\subset \c{A}$ denote by $\interior_{\c{A}}S$, $\closure_{\c{A}}S$, $\interior_{\c{X}}S$, and $\closure_{\c{X}}S$ its interior and closure in $\c{A}$ and $\c{X}$, respectively.
Show that
$\interior_{\c{A}}S\supset \interior_{\c{X}}S$ and $\closure_{\c{A}}S\subset \closure_{\c{X}}S$.
Provide examples showing that these inclusions might be strict.
\end{thm}

Let $A$ be a subset of a topological space $\c{X}$ and let $\c{Y}$ be another topological space.
A map $h\:A\to \c{Y}$ is called \index{continuous map}\emph{continuous} if it is continuous with respect to the induced topology on $A$.

Recall the \index{restriction}\emph{restriction} of a map $f:\c{X}\to \c{Y}$ to $A\subset\c{X}$ is obtained by keeping the same rule as $f$ but only allowing inputs from~$A$.
It is usually denoted by $f|_A$; so $(f|_A)(a)=f(a)$ if $a\in A$ and otherwise it is undefined.

Observe that $(f|_A)^{-1}(W)=A\cap f^{-1}(W)$ for any $W\subset \c{Y}$.
It follows that if $f$ is continuous, then so is $f|_A\:A\to\c{Y}$;
here we consider set $A$ with the induced topology.
The following exercise provides a partial converse.

\begin{thm}{Exercise}\label{ex:u-continuous}
Let $f\:\c{X}\to \c{Y}$ be a map between topological spaces.
Suppose $A,B\subset\c{X}$ are sets such that $A\cup B=\c{X}$ and either
\begin{subthm}{ex:u-continuous:open}
both $A$ and $B$ are open sets, or
\end{subthm}
\begin{subthm}{ex:u-continuous:closed}
both $A$ and $B$ are closed sets.
\end{subthm}
Show that $f$ is continuous if and only if so are the restrictions $f|_A$ and $f|_B$.

\begin{subthm}{ex:u-continuous:none}
Show that this is no longer true without assumptions of \ref{SHORT.ex:u-continuous:open} or~\ref{SHORT.ex:u-continuous:closed}.
\end{subthm}

\end{thm}

\section{Neighborhoods}

Let $x$ be a point in a topological space $\c{X}$.
A \index{neighborhood}\emph{neighborhood} of $x$ is any open set $N$ containing $x$.

In topology, neighborhoods often replace balls, which makes sense only in metric spaces.

\begin{thm}{Exercise}\label{ex:bry-nbhd}
Let $A$ be a set in a topological space $\c{X}$.
Show that $x\in \partial A$ if and only if any neighborhood of $x$ contains points of $A$ and  of its complement $\c{X}\setminus A$.
\end{thm}

Let $A$ and $B$ be subsets of a topological space $\c{X}$.
The set $A$ is said to be \index{dense set}\emph{dense} in $B$ if $\bar A\supset B$;
if $\bar A=\c{X}$, then we say that $A$ is \index{everywhere dense}\emph{everywhere dense}.

\begin{thm}{Exercise}\label{ex:dense-nbhd} 
Show that $A$ is dense in $B$ if and only if any neighborhood of any point in $B$ intersects $A$.
\end{thm}

\section{Limits}\label{sec:limits}

\begin{thm}{Definition}\label{def:limit-top}
Suppose $x_n$ is a sequence of points in a topological space $\c{X}$.
We say that $x_n$ \index{convergence}\emph{converges} to a point $x_\infty\in \c{X}$ (briefly $x_n\to x_\infty$ as $n\to\infty$) if for any neighborhood $N$ of $x_\infty$, we have that 
$x_n\in N$ for all sufficiently large $n$.
\end{thm}

Observe that the above definition agrees with \ref{def:limit-metric}.
In other words, a sequence of points $x_1,x_2,\dots$ in a metric space converges to a point $x_\infty$ in the sense of the definition \ref{def:limit-metric} if and only if it converges in the sense of the definition \ref{def:limit-top}.

\begin{thm}{Exercise}\label{ex:lim-weaker}
Show that a convergent sequence of points in a topological space is also convergent for every weaker topology.
\end{thm}

Note that in a space with the concrete topology any sequence converges to any point.
In particular, a sequence might have several different limits.
Furthermore, if we equip $\RR$ with the cofinite topology then any sequence of pairwise distinct numbers converges to every point in $\RR$.

The following exercise shows that converging sequences do \textit{not} adequately describe the topology of a space;
namely, an analog of \ref{ex:continuous-limit} does not hold.\footnote{So-called \index{net}\emph{nets} \cite{mcshane} provide an appropriate generalization of sequences that works well in topological spaces, but we are not going to consider them.}

Recall that a set is called \index{countable set}\emph{countable} if it admits a bijection to a subset of the set of natural numbers.
In particular, all finite sets are countable.

\begin{thm}{Advanced exercise}\label{ex:cocountable-top}
Let $\c{X}$ be the set of real numbers with the so-called \index{cocountable topology}\emph{cocountable topology};
its closed sets are either countable or the whole $\RR$.

\begin{subthm}{}
Show that a sequence $x_1,x_2,\dots \in \c{X}$ converges to $x_\infty$ if and only if $x_n=x_\infty$ for all sufficiently large $n$.
\end{subthm}

\begin{subthm}{ex:cocountable-top:closed}
Show that any set $A\subset \c{X}$ is \index{sequentially closed set}\emph{sequentially closed};
that is, if $a_1,a_2,\dots \in A$ is a converging sequence, then its limit lies in $A$.
Observe that not all subsets of $\c{X}$ are closed.
\end{subthm}

\begin{subthm}{}
Show that any map $f\:\c{X}\to \c{X}$ is \index{sequentially continuous map}\emph{sequentially continuous};
that is, if a sequence $x_1,x_2,\dots \in \c{X}$ converges to $x_\infty$ in $\c{X}$, then  $y_n=f(x_n)$ converges to $y_\infty\z=f(x_\infty)$.
Observe that not all maps $f\:\c{X}\to \c{X}$ are continuous.
\end{subthm}

\end{thm}
