\chapter{Separation theorem}\label{chap:Jordan}

The Jordan separation theorem states that the complement of a closed simple curve in the plane is not connected.
This theorem is known for its simple formulation and annoyingly tricky proofs.
We will present the proof of Patrick Doyle \cite{doyle} that is based on Klee's trick.
This proof is among the shortest; it also uses several results in topology, which are worth knowing.

\section{Tietze--Urysohn extension theorem}

The following statement is a partial case of the so-called \index{Tietze--Urysohn extension theorem}\emph{Tietze--Urysohn extension theorem},
which provides an analogous statement for all normal spaces, which means $T_1+T_4$; see \ref{sec:haudorff-comments}.

\begin{thm}{Proposition}\label{thm:tietze}
Let $\Gamma$ be a closed subset of $\RR^2$.
Then any continuous function $f\:\Gamma\to\RR$ admits a continuous extension to $\RR^2$;
that is, there is a continuous function $F\:\RR^2\to\RR$ such that $F(p)=f(p)$ for any point $p\in\Gamma$.
\end{thm}

\parit{Proof.}
We can assume that $\Gamma\ne\emptyset$, otherwise take $F(x)\equiv 0$.

Assume $f(\Gamma)\subset(1,2)$.
Given $x\in\RR^2$, let
\[
a(x)=\inf\limits_{y\in\Gamma} \{\,|x-y|\,\}
\quad\text{and}\quad
b(x)=\inf\limits_{y\in\Gamma} \{\,f(y)\cdot |x-y|\,\}.
\]
Observe that
\[a(x)<b(x)<2\cdot a(x)\]
for any $x\notin \Gamma$.
Therefore
\[F(x)=
\begin{cases}
f(x)& \text{if}\ x\in \Gamma,
\\
\frac{b(x)}{a(x)}& \text{if}\ x\notin \Gamma
\end{cases}
\]
defines a function $F\:\RR^2\to(1,2)$.

It remains to check that $F$ is continuous;
that is, if a converging sequence of points $p_n\to p_\infty$, then $F(p_\infty)=\lim F(p_n)$.

We can assume that $\lim F(p_n)$ is defined, otherwise pass to a subsequence.
Furthermore, by the triangle inequality, $a$ and $b$ are continuous functions.
Since $a(x)>0$ for any $x\notin\Gamma$, the function $F$ is continuous in the complement $\RR^2\setminus \Gamma$.

It remains to check the case $p_\infty\in\Gamma$.
Furthermore, if $p_n\in \Gamma$ for all large $n$, then the statement follows from the continuity of $f$.
Therefore, passing to a subsequence, we can assume that $p_n\notin \Gamma$ for any $n$.

Let $q_n,s_n\in\Gamma$ be points that minimize $|p_n-q_n|$ and $f(s_n)\cdot |p_n-s_n|$, respectively;
so
\[
\begin{aligned}
a(p_n)&=|p_n-q_n|,
\\
b(p_n)&=f(s_n)\cdot |p_n-s_n|,
\\
F(p_n)&=f(s_n)\cdot \frac{|p_n-q_n|}{|p_n-s_n|}.
\end{aligned}
\eqlbl{eq:abF}
\]
The existence of $q_n$ and $s_n$ follows since $\dist_x$ and $f$ are continuous on $\Gamma$ and $\Gamma\cap \cBall[x,R]$ is a nonempty compact set for large $R$.

Note that
\[\tfrac12\cdot|p_n-s_n| \le |p_n-q_n|\le |p_n-p_\infty|;\]
in particular, $q_n\to p_\infty$, $s_n\to p_\infty$, and $f(s_n)\to f(p_\infty)$ as $n\to\infty$.
Let
\[\eps_n=\sup\set{|f(p_\infty)-f(p)|}{p\in \Gamma\cap \cBall(p_\infty,2\cdot |p_n-p_\infty|};\]
note that $\eps_n\to 0$ as $n\to\infty$ and
\[1-2\cdot \eps_n\le \frac{|p_n-q_n|}{|p_n-s_n|}\le 1\]
It follows that $\tfrac{|p_n-q_n|}{|p_n-s_n|}\to1$ as $n\to \infty$, and by \ref{eq:abF}, continuity of $F$ follows.

In general case, let $\hat f=h\circ f\:\Gamma\to (1,2)$ for a homeomorphism $h\:\RR\to (1,2)$.
If $\hat F\:\RR^2\to (1,2)$ is extension of $\hat f$, then $F\z=h^{-1}\circ \hat F$ is the required extension of $f$.
\qeds

\section{Around Van Kampen}

\begin{thm}{Lemma}\label{prop:c/sc}
Let us identify $\RR^2$ with the $(x,y)$-plane in $\RR^3$.

Suppose $\Gamma\subset\RR^2$ is a closed set.
Then the complement $X=\RR^2\setminus \Gamma$ is connected if and only if the complement $Y=\RR^3\setminus \Gamma$ is simply-connected.
\end{thm}


The proof is based on \ref{prop:van-kamp:sc} and \ref{ex:vankam-nonconnect}.


\parit{Proof of \ref{prop:c/sc}.}
Denote by $\Gamma_{-}$ (respectively $\Gamma_{+}$) the sets that include $\Gamma$ and the points below (respectively above) $\Gamma$;
that is,
\begin{align*}
\Gamma_{-}&=\set{(x,y,z)}{(x,y)\in\Gamma\ \text{and}\ z\le 0},
\\
\Gamma_{+}&=\set{(x,y,z)}{(x,y)\in\Gamma\ \text{and}\ z\ge 0}.
\end{align*}

Consider the complements $V=\RR^3\setminus \Gamma_{-}$ and $W=\RR^3\setminus \Gamma_{+}$.
Note that $X\times \RR= V\cap W$ and $Y=V\cup W$.

The sets $V$ and $W$ are contractible; in particular, simply-connected.
Indeed, the horizontal plane $z= 1$ is a deformation retract of $V$;
a retraction can be defined by $(x,y,z)\mapsto (x,y,1)$ and the following homotopy shows that it is homotopic to the identity map:
\[h_t(x,y,z)=(x,y,(1-t)\cdot z+t).\]
The plane $z= 1$ is contractible, hence so is $V$.
Similarly, $W$ is contractible since it has plane $z=-1$ as a deformation retract.

If $X$ is not connected, then so is $X\times \RR= V\cap W$.
In this case, by \ref{ex:vankam-nonconnect}, we have $\RR^3\setminus \Gamma=V\cup W$ is not simply-connected.

On the other hand if $X$ is connected, then so is $X\times \RR= V\cap W$.
In this case, by \ref{prop:van-kamp:sc}, we have $\RR^3\setminus \Gamma=V\cup W$ is simply-connected.
\qeds

\section{Klee trick}

\begin{thm}{Proposition}\label{prop:home-gamma-z-axis}
Suppose $\Gamma\subset\RR^2$ is a closed subset homeomorphic to $\RR$.
Then there is a homeomorphism $\RR^3\selfmap$ that maps $\Gamma$ to the $z$-axis.
\end{thm}

The following proof uses the so-called \index{Klee trick}\emph{Klee trick}, which is quite useful in many topological problems.
In fact, the construction gives a bit more; it implies the existence of a one-parameter family of homeomorphisms $h_t\:\RR^3\selfmap$
that starts at the identity map $\id_{\RR^3}$
and ends at the homeomorphism we need.

\parit{Proof.}
Let $h\:t\mapsto (a(t),b(t))$ be a homeomorphism $\RR\to \Gamma$.
Applying \ref{thm:tietze} to $f=h^{-1}$, we get a function $F\:\RR^2\z\to\RR$ such that
\[F(a(t),b(t))=t\]
for any $t\in \RR$.

Note that the map
\[p\:(x,y,z)\mapsto (x,y,z+F(x,y))\] is a homeomorphism.
Indeed, this map is continuous and its inverse
\[p^{-1}\:(x,y,z)\z\mapsto (x,y,z-F(x,y))\]
is continuous as well.

Similarly, the map
\[q\:(x,y,z)\mapsto (x-a(z),y-b(z),z)\]
is a homeomorphism as well.
Indeed, $q$ is continuous and it has an inverse
\[q^{-1}\:(x,y,z)\mapsto (x+a(z),y+b(z),z)\]
that is continuous as well.

It follows that the composition $q\circ p\:\RR^3\to\RR^3$ is a homeomorphism.
Since $F(a(t),b(t))=t$ for any $t$, we have
\[q\circ p(a(t),b(t),0)=q(a(t),b(t),t)=(0,0,t).\]
It follows that $q\circ p$ sends $\Gamma$ to the $z$-axis as required.
\qeds

\begin{thm}{Exercise}\label{ex:klee[0,1]-1}

Construct a homeomorphism $h\:\RR^3\to \RR^3$ that maps $I$ to a line segment.
\end{thm}

\section{Separation by a proper curve}

\begin{thm}{Theorem}\label{thm:proper-jordan}
Suppose $\Gamma\subset\RR^2$ is a closed set homeomorphic to~$\RR$.
Then $\RR^2\setminus \Gamma$ has at least two connected components.
\end{thm}

Note that the assumption that $\Gamma$ is closed is necessary;
indeed, a finite open interval $I$ of a line in $\RR^2$ is homeomorphic to $\RR$, but its complement $\RR^2\setminus I$ is connected.

\parit{Proof.}
Assume $\RR^2\setminus \Gamma$ is connected.
Then by \ref{prop:c/sc}, $\RR^3\setminus \Gamma$ is simply-connected.

On the other hand by \ref{prop:home-gamma-z-axis}, $\RR^3\setminus \Gamma$ is homeomorphic to the complement of $z$-axis in $\RR^3$,
which is not simply-connected since it can be retracted to a circle%
\footnote{Alternatively one may apply \ref{prop:c/sc} again.} --- a contradiction.
\qeds

\begin{thm}{Exercise}\label{ex:connected-complement}
Let $A$ and $B$ be disjoint closed subsets in $\RR^2$ such that both complements $\RR^2\setminus A$ and $\RR^2\setminus B$ are connected.
Show that $\RR^2\setminus(A\cup B)$ is connected.
\end{thm}

\begin{thm}{Exercise}\label{ex:jordanI+}

Show that $\RR^2\setminus I$ is connected.
\end{thm}

\begin{thm}{Exercise}\label{ex:proper-jordan3}

\end{thm}

\section{Jordan's theorem}

\begin{thm}{Theorem}\label{thm:jourdan}
If $J\subset\mathbb{S}^2$ is a subset homeomorphic to $\mathbb{S}^1$,
then $\mathbb{S}^2\setminus J$ has at least two connected components.
\end{thm}


\parit{Proof.}
Remove a point $p$ from $J$ to get a closed set $\Gamma=J\setminus\{p\}$ homeomorphic to $\RR$ in $\mathbb{S}^2\setminus\{p\}\simeq\RR^2$.
It remains to apply \ref{thm:proper-jordan}.
\qeds

Note that applying \ref{ex:proper-jordan3} in the proof of \ref{thm:jourdan}, one gets that
$\mathbb{S}^2\setminus J$ has exactly two connected components.
Furthermore, this theorem has a yet more exact version, which is due to Arthur Schoenflies;
it states that there is a homeomorphism $h\:\SS^2\selfmap$ that maps $J$ to the equator;
in particular, each connected component of $\mathbb{S}^2\setminus J$ is homeomorphic to an open disc.
