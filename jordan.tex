\chapter{Jordan separation theorem}

The Jordan separation theorem states that the complment of a closed simple curve in the plane is not connected.
This theorem is known for formulation and annoyingly tricky proofs.
We will present the proof of Patrick Doyle \cite{doyle};
it is among the shortest proofs.
This proof use several results in topology that, which worth knowing.

\section{Tietze--Urysohn extension theorem}

The following statement is a partial case of the so-called \index{Tietze--Urysohn extension theorem}\emph{Tietze--Urysohn extension theorem}.

\begin{thm}{Proposition}\label{thm:tietze}
Let $\Gamma$ be a closed subset in $\RR^2$ and let $\II$ be an open intevel.
Then any continuous function $f\:\Gamma\to\II$ admits a continuous extension to $\RR^2$;
that is, there is a continuous function $F\:\RR^2\to\II$ such that $F(p)=f(p)$ for any point $p\in\Gamma$.
\end{thm}


\parit{Proof.}
Assume $\II=(1,2)$.
Given $x\in\RR^2$, let
\[
a(x)=\inf\limits_{y\in\Gamma} \{\,|x-y|\,\}
\quad\text{and}\quad
b(x)=\inf\limits_{y\in\Gamma} \{\,f(y)\cdot |x-y|\,\}.
\]
Observe that
\[F(x)=
\begin{cases}
f(x)& \text{if}\ x\in \Gamma,
\\
\frac{b(x)}{a(x)}& \text{if}\ x\notin \Gamma
\end{cases}
\]
is the needed extension.

In general case, choose a homeomorphism $h\:\II\to (1,2)$.
Let $\hat f=h\circ h\:\Gamma\to (1,2)$ and let $\hat F\:\Gamma\to (1,2)$ be its extension.
Then $F\z=h^{-1}\circ \hat F$ is the requred extension of $f$.
\qeds

\section{Klee trick}

\begin{thm}{Proposition}\label{prop:home-gamma-z-axis}
Suppose $\Gamma\subset\RR^2$ is a closed subset homeomorphic to $\RR$.
Then there is a homeomorphism $\RR^3\to \RR^3$ that maps $\Gamma$ to the $z$-axis.
\end{thm}

The following proof uses the so-called \index{Klee trick}\emph{Klee trick}, which is quite useful in many topological problems.

One can extract more from the proof, namely there is a continous map $(x,t)\mapsto h_t(x)$ from $\RR^3\times[0,1]\to\RR^3$ such that $h_t\:\RR^3\to\RR^3$ is a homeomorphism for any $t\in [0,1]$; furthermore, $_0=\id_{\RR^3}$ and $h_1$ maps $\Gamma$ to the $z$-axis.

\parit{Proof of \ref{prop:home-gamma-z-axis}.}
Let $h\:t\mapsto (a(t),b(t))$ be a homeomorphism $\RR\to \Gamma$.
Applying \ref{thm:tietze} to $f=h^{-1}$, we get a function $F\:\RR^2\z\to\RR$ such that
\[F(a(t),b(t))=t\]
for any $t\in \RR$.

Note that the map
\[A\:(x,y,z)\mapsto (x,y,z+f(x,y))\] is a homeomorphism.
Indeed, this map is continuous and its inverse
\[A^{-1}\:(x,y,z)\z\mapsto (x,y,z-f(x,y))\]
is continuous as well.

Similarly, the map
\[B\:(x,y,z)\mapsto (x-a(z),y-b(z),z)\]
is a homeomorphism as well.
Indeed, $B$ is continuous and it has an inverse
\[B^{-1}\:(x,y,z)\mapsto (x+a(z),y+b(z),z)\]
that is continuous as well.

It follows that the composition $B\circ A\:\RR^3\to\RR^3$ is a homeomorphism.
Since $f(a(t),b(t))=t$,
\[B\circ A(a(t),b(t),0)=B(a(t),b(t),t)=(0,0,t).\]
It follows that $B\circ A$ sends $\Gamma$ to the $z$-axis as required.
\qeds

\section{Separation by a proper curve}

\begin{thm}{Theorem}\label{thm:proper-jordan}
Suppose $\Gamma\subset\RR^2$ is a closed set homeomorphic to~$\RR$.
Then $\RR^2\setminus \Gamma$ has at least two connected components.
\end{thm}

Note that the assumption that $\Gamma$ is closed is necessary;
indeed, a finite open interval $I$ of a line in $\RR^2$ is homeomorphic to $\RR$, but its complement $\RR^2\setminus I$ is connected.

The theorem follows from \ref{prop:c/sc}, \ref{prop:home-gamma-z-axis}, and \ref{ex:z-axis}.


\parit{Proof of \ref{prop:c/sc}.}
Denote by $A$ (respectively $B$) the sets that include $\Gamma$ and the points below (respectively above) $\Gamma$;
that is,
\begin{align*}
A&=\set{(x,y,z)}{(x,y)\in\Gamma\ \text{and}\ z\le 0},
\\
B&=\set{(x,y,z)}{(x,y)\in\Gamma\ \text{and}\ z\ge 0}.
\end{align*}
Consider their complements $V=\RR^3\setminus A$ and $W=\RR^3\setminus B$.
Note that $Y=V\cup W$.

The sets $V$ and $W$ are simply-connected.
Indeed, the horizontal plane $z= 1$ is a deformation retract of $V$;
a retraction can be defined by $(x,y,z)\mapsto (x,y,1)$ and the following homotopy shows that it is homotopic to the identity map: 
\[h_t(x,y,z)=(x,y,(1-t)\cdot z+t).\]
The plane is contractible, in particular simply-connected; therefore so is $V$.
Similarly, one proves that $W$ is simply-connected.

Since $X$ is an open and connected set in $\RR^2$,
by \ref{thm:open-connected=path-connected}, $X$ is path-connected.
Further, note that $V\cap W=X\times \RR$.
Therefore $V\cap W$ is path-connected as well.

Summarizing, $V$ and $W$ are open simply-connected sets, $Y=V\z\cup W$, and $V\z\cap W$ is path-connected.
Applying \ref{ex:sc-VuW}, we get that $Y$ is simply-connected.
\qeds

\section{Jordan's theorem}

\begin{thm}{Theorem}
If $J\subset\mathbb{S}^2$ be a subset homeomorphic to $\mathbb{S}^1$,
then $\mathbb{S}^2\setminus J$ has at least two connected components.
\end{thm}


\parit{Proof.}
Remove a point $p$ from $J$ to get a closed line $\Gamma=J\setminus\{p\}$ in $ \mathbb{S}^2\setminus\{p\}\simeq\RR^2$.
It remains to apply \ref{thm:proper-jordan}.
\qeds

\begin{thm}{Exercise}\label{ex:connected-complement}

Show that $\RR^2\setminus(A\cup B)$ is connected.
\end{thm}
