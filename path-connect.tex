\chapter{Path-connected spaces}\label{chap:path-connected}

\section{Paths}\label{sec:Operations on paths}

Let $\c{X}$ be a topological space.
A continuous map $a\:[0,1]\to \c{X}$ is called a \index{path}\emph{path}.
If $x=a(0)$ and $y=a(1)$ we say that $a$ is a path from $x$ to $y$.

Let us start with several examples and constructions of paths.
\begin{itemize}
\item A path $a$ is called \index{constant path}\emph{constant} if it stays at one point.
In other words, $a(t)=p$ for some fixed point $p$; in this case, the path $a$ can be denoted by $e_p$.

\item Given a path $a\:[0,1]\to \c{X}$, one can consider the \index{time-reversed path}\emph{time-reversed} path $\bar a$, defined by
\[\bar a(t)=a(1-t).\]
Note that $\bar a$ is continuous since $a$ is.

\item Let $a$ and $b$ be paths such that $a(1)=b(0)$.
Then we can join them into one $c\:[0,1]\to \c{X}$ defined as
\[c(t)=
\left\{
\begin{aligned}
a(2\cdot t)&&\text{if}&\ t\le \tfrac12,
\\
b(2\cdot t-1)&&\text{if}&\ t\ge \tfrac12;
\end{aligned}
\right.
\]
so, $c$ follows $a$ and then $b$.
The path $c$ is called the \index{product of paths}\emph{product} (or \index{concatenation}\emph{concatenation}) of $a$ and $b$; it is denoted as $c=a * b$.
Note that \ref{ex:u-continuous:closed} implies that $a*b$ is continuous;
in other words $a*b$ is indeed a path.
\end{itemize}

\begin{thm}{Advanced exercise}\label{ex:simple-path}
Let $a$ be a path from $x$ to $y$ in a Hausdorff space $\c{X}$.
Assume $x\ne y$.
Show that there is an injective path $b$ from $x$ to $y$ such that $b([0,1])\subset a([0,1])$.
\end{thm}

\section{Path-connected components}\label{sec:Path-connected components}

Consider the following relation on the set of points of a topological space:
\[x\sim y \quad\Leftrightarrow\quad\text{there is a path from $x$ to $y$.}\]

From \ref{sec:Operations on paths}, $\sim$ is an equivalence relation on the set of points in $\c{X}$; in other words, the following properties hold for any points $x,y,z\in \c{X}$:
\begin{itemize}
 \item $x\sim x$ (since $e_x$ is a path from $x$ to $x$);
 \item if $x\sim y$, then $y\sim x$ (indeed, if $a$ is a path from $x$ to $y$, then $\bar a$ is a path from $y$ to $x$);
 \item if $x\sim y$ and $y\sim z$, then $x\sim z$ (indeed, if $a$ is a path from $x$ to $y$,
 and $b$ is a path from $y$ to $z$ then $a*b$ is a path from $x$ to $z$).
\end{itemize}

The equivalence class of a point $x$ for the equivalence relation $\sim$ is called the \index{path-connected component}\emph{path-connected component} of $x$;
this is the set of all points in $\c{X}$ that can be joined with $x$ by a path.


\section{Path-connected spaces}\label{sec:flea and comb}

A space $\c{X}$ is called \index{path-connected space}\emph{path-connected} if it is nonempty and any two points in $\c{X}$ can be connected by a path;
that is, for any $x,y\in \c{X}$ there is a path $a$ from $x$ to $y$.
One can say that $\c{X}$ is path-connected if $\c{X}$ is a path-connected component of one (and therefore every) point $x\in\c{X}$.

\begin{thm}{Theorem}\label{thm:conncted/path-connected}
Any path-connected space is connected;
the converse does not hold.
\end{thm}

\parit{Proof; main part.}
Let $\c{X}$ be a path-connected space.

By \ref{prop:connected[0,1]}, the unit interval $[0,1]$ is connected.
By \ref{prop:image-connected}, for any path $a\:[0,1]\to \c{X}$ the image $a([0,1])$ is connected.

Fix $x\in \c{X}$.
Since $\c{X}$ is path-connected,
\[\c{X}=\bigcup_a a([0,1]),\]
where the union is taken for all paths $a$ starting from $x$.
It remains to apply~\ref{prop:union-connect}.

\parit{Second part.}
We need to present an example of a space that is connected, but not path-connected.

\end{wrapfigure}

Denote by $I$ the closed line segment from $(0,0)$ to $(1,0)$ in $\RR^2$.
Further, denote by $J_n$ the closed line segment from $(\tfrac1n,0)$ to $(\tfrac1n,1)$.
Consider the union of all these segments
\[W=I\cup J_1\cup J_2 \cup \ldots\]
and set 
\[W'=W\cup\{y\},\]
where $y=(0,1)$.
The space $W'$ is called the {}\emph{flea and comb}; %??? W' >>> \dot W???
the set $W$ is called the \index{comb}\emph{comb},
and the point $y$ is called the \index{flea}\emph{flea}.

Note that $W\subset W'\subset \bar W$.
Therefore, by \ref{ex:A<B<bar-A}, $W'$ is connected.

It remains to show that $W'$ is not path-connected.
Assume the contrary. 
Let $a$ be a path from $x=(0,0)$ to $y=(0,1)$.

Note that $a^{-1}(\{y\})$ is a closed subset of the compact space $[0,1]$.
Therefore $a^{-1}(\{y\})$ is compact (\ref{prop:compact-closed}).
In particular, the set $a^{-1}(\{y\})$ has the minimum element,
denote it by $t_1$.
Note that $t_1>0$, $a(t_1)=y$ and $a(t)\ne y$ for any $t<t_1$.

Since $a$ is continuous, there is $t_0<t_1$ such that $|a(t)-y|<\tfrac1{10}$ for any $t\in [t_0,t_1]$.
Since $|a(t_0)-y|<\tfrac1{10}$, we have $a(t_0)\in J_n$ for some $n$ (in other words, $a(t_0)\notin I$).

Denote by $N$ the intersection of the $\tfrac1{10}$-neighborhood of $y$ with $W$.
The intersection of $J_n\cap N$ is a connected component of $N$.
By \ref{prop:image-connected}, $a(t)\in J_n$ for any $t\in [t_0,t_1]$;
in particular, $a(t_1)\ne y$ --- a contradiction.
\qeds

\section{Locally path-connected spaces}\label{sec:Locally path-connected spaces}

A topological space $\c{X}$ is called \index{locally path-connected}\emph{locally path-connected} if for any point $p\in\c{X}$ and any open set $V\ni p$ there is a path-connected open set $W$ such that $V\supset W\ni p$.
For instance, Euclidean space is locally path-connected;
it follows since any open ball in a Euclidean space is convex, and therefore path-connected.

\begin{thm}{Theorem}\label{thm:open-connected=path-connected}
A connected open set in a locally path-connected space is path-connected.
\end{thm}

\parit{Proof.}
Let $V$ be an open subset of a locally path-connected space.
Choose a point $p\in V$;
denote by $P\subset V$ the path-connected component of $p$ in $V$.

Choose $q\in V$ and let $W_q$ be the neighborhood provided by the definition of locally path-connected space;
that is, $V\supset W_q\ni q$.
Since $W_q$ is path-connected, we have $W_q\subset P$ or $W_q\cap P=\emptyset$.

It follows that $P$ and its complement $V\setminus P$ are open.
Since $V$ is connected, we get that $V\setminus P=\emptyset$ --- hence the result.
\qeds
