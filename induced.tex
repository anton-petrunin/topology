\chapter{Constructions}\label{chap:constructions}

Here we introduce several constructions that produce new topological spaces from the given ones.

\section{Product space}\label{sec:Product space}

Recall that $\c{X}\times\c{Y}$ denotes the set of all pairs $(x,y)$ such that $x\in\c{X}$ and  $y\in\c{Y}$.

Suppose that the sets $\c{X}$ and $\c{Y}$ are equipped with topologies.
Let us equip $\c{X}\times\c{Y}$ with topology by declaring that a set is open in $\c{X}\times\c{Y}$ if it can be presented as a union of sets of the following type: $V\times W$ for open sets $V\subset \c{X}$ and $W\subset \c{Y}$.
In other words, a subset $O$ is open in $\c{X}\times\c{Y}$ if and only if there are familys of open sets $V_\alpha\subset \c{X}$ and $W_\alpha\subset \c{Y}$  such that
\[O=\bigcup_\alpha V_\alpha\times W_\alpha,\]
here $\alpha$ runs in some index set.

By default, we assume that $\c{X}\times\c{Y}$ is equipped with the product topology;
in this case, $\c{X}\times\c{Y}$ is called the \index{product space}\emph{product space}.

\begin{thm}{Proposition}
The product topology is indeed a topology.
\end{thm}

\parit{Proof.}
Parts \ref{SHORT.def:top-space:empty} and \ref{SHORT.def:top-space:u} in \ref{def:top-space} are evident.
It remains to check \ref{SHORT.def:top-space:n}.
Consider two sets
\[
O=\bigcup_\alpha V_\alpha\times W_\alpha
\quad\text{and}\quad
O'=\bigcup_\beta V'_\beta\times W'_\beta.
\]
where $\alpha$ and $\beta$ run in some index sets, say $\c{I}$ and $\c{J}$ respectively.
We need to show that $O\cap O'$ can be presented as a union of products of open sets;
the latter follows from the next set-theoretical identity
\[O\cap O'=\bigcup_{\alpha,\beta} (V_\alpha\cap V'_\beta)\times (W_\alpha\cap W'_\beta).\eqlbl{eq:nu=un}\]

Checking  \ref{eq:nu=un} is straightforward.
Indeed, $(x,y)\in O\cap O'$ means that $(x,y)\in O$ and $(x,y)\in O'$;
the latter means that $x\in V_\alpha$, $y\in W_\alpha$ and $x\in V'_\beta$, $y\in W'_\beta$ for \textit{some} $\alpha$ and $\beta$.
In other words, $x\in V_\alpha\cap V'_\beta$ and $y\in W_\alpha\cap W'_\beta$ for \textit{some} $\alpha$ and $\beta$;
the latter means that $(x,y)$ belongs to the right-hand side in \ref{eq:nu=un}.
\qeds

By spelling out the above definition, we get the following.

\begin{thm}{Observation}
Two maps $f\:\c{X}\to\c{Y}$ and $g\:\c{X}\to\c{Z}$ are continuous if and only if
$x\mapsto (f(x),g(x))$ is a continuous map $\c{X}\to\c{Y}\times \c{Z}$.
\end{thm}


\begin{thm}{Exercise}\label{ex:cont-product-constr}
Let $f$ and $g$ be continuous real-valued functions on the topological space $\c{X}$.
Show that (a) $h_1=f+g$, (b) $h_2=f\cdot g$, and (c) $h_3(x)\df\max\{\,f(x), g(x)\,\}$ are continuous.

\end{thm}

\begin{thm}{Exercise}\label{ex:bryAxB}
Show that
$\partial (A\times B)=(\partial A\times \bar B)\cup(\bar A\times \partial B)$
for any subsets $A\subset\c{X}$, $B\subset\c{Y}$ and their product $A\times B$ in the product space $\c{X}\times \c{Y}$.
\end{thm}


%\begin{thm}{Exercise}\label{ex:[0,1)x[0,1)} Show that $[0,1)\times [0,1)$ is homeomorphic to $[0,1]\times [0,1)$.

\begin{thm}{Exercise}\label{ex:cont-product}
Construct a function $f\:[0,1]\times [0,1]\to [0,1]$ such that $f$ is not continuous, but the functions
$x\mapsto f(a,x)$ and $x\mapsto f(x,b)$ are continuous for any $a,b\in [0,1]$.
\end{thm}


\section{Base}

\begin{thm}{Definition}\label{def:base}
A family $\s{B}$ of open sets in a topological space $\c{X}$ is called its \index{base}\emph{base} if every open set in $\c{X}$ is a union of sets in $\s{B}$.
\end{thm}

By \ref{ex:union-of-balls}, \textit{open balls form a base of a metric space}.

A base completely defines its topology,
but typically a topology has many different bases.
In metric spaces, for example, the set of all balls with rational radii is a base;
another example is the set of all balls with radii smaller than $1$.

In many cases, it is convenient (and also economical) to describe the topology by specifying its base.
For example, the product topology on $\c{X}\times\c{Y}$ can be redefined as a \textit{topology with a base formed by all products $V\times W$, where $V$ is open in $\c{X}$, and $W$ is open in $\c{Y}$}.

\begin{thm}{Exercise}\label{ex:base-nbhd}
Let $\s{B}$ be a family of open sets in a topological space $\c{X}$.
Show that $\s{B}$ is a base in $\c{X}$ if and only if for any point $x\in \c{X}$ and any neighborhood $N\ni x$ there is $B\in\s{B}$ such that $x\in B\subset N$.
\end{thm}

\begin{thm}{Proposition}\label{prop:base-any}
Let $\s{B}$ be a set of subsets of some set $\c{X}$.
Then $\s{B}$ is a base of some topology on $\c{X}$ if and only if it satisfies the following conditions:

\begin{subthm}{prop:base-any:covers}
$\s{B}$ {}\emph{covers} $\c{X}$;
that is, every point $x\in\c{X}$ lies in some set $B\in \s{B}$.
\end{subthm}

\begin{subthm}{prop:base-any:n}
For each pair of sets $B_1, B_2\in \s{B}$ and each point $x \in B_1 \cap B_2$ there exists a set $B
\in \s{B}$ such that $x\in B\subset B_1 \cap B_2$.
\end{subthm}

\end{thm}

\parit{Proof.}
Denote by $\s{O}$ the set of all unions of sets in $\s{B}$.
We need to show that $\s{O}$ is a topology on $\c{X}$.

Evidently, the union of any family of sets in $\s{O}$ is in $\s{O}$.
Further, $\c{X}$ is in $\s{O}$ by \ref{SHORT.prop:base-any:covers}.
The empty set is in $\s{O}$ since it is a union of the empty family.

It remains to check \ref{def:top-space:n}; suppose
\[O=\bigcup_\alpha B_\alpha
\quad\text{and}\quad
O'=\bigcup_\beta B'_\beta,
\]
where $\alpha$ and $\beta$ run in some index sets, and $B_\alpha$, $B'_\beta\in \s{B}$ for any $\alpha$ and~$\beta$.
Then $x\in O\cap O'$ if and only if for some $\alpha$ and $\beta$ we have $x\in B_\alpha$ and $x\in B'_\beta$.
By \ref{SHORT.prop:base-any:n}, we can choose $B\in \s{B}$ so that $x\in B\subset B_\alpha \cap B'_\beta$.
Since $B_\alpha \cap B'_\beta\subset O\cap O'$, it follows that 
\[\textit{for any $x\in O\cap O'$ there is $B_x\in\s{B}$ such that 
$x\in B_x\subset O\cap O'$.}\]
Observe that 
\[O\cap O'=\bigcup_{x\in O\cap O'} B_x.\]
It follows that $O\cap O'\in\s{O}$ if $O,O'\in\s{O}$.
\qeds

\begin{thm}{Exercise}\label{ex:base-N}
Let $\NN=\{\,1,2,\dots\,\}$ be the set of natural numbers,
and let $\s{B}$ be a set of all arithmetic progressions in $\NN$;
that is, $\s{B}$ includes the set $\{\,a,a+d,a+2\cdot d,\dots\,\}$ for any $a, d\in \NN$.

Show that $\s{B}$ is a base of some topology on $\NN$.
Is it true that in this topology the set $\{1\}$ is open and/or closed?

\end{thm}


\section{Prebase}\label{prebase}

Suppose $\s{P}$ is a family of subsets of $\c{X}$ that covers the whole space;
that is, $\c{X}$ is a union of all sets in $\s{P}$.
By \ref{prop:base-any}, the set of all finite intersections of sets in $\s{P}$ is a base for \textit{some} topology on $\c{X}$.
The set $\s{P}$ is called a \index{prebase}\emph{prebase} for this topology (also known as a \index{subbase}\emph{subbase}).

Note that any base is a prebase, but not the other way around.
Spelling the defining properties of open sets (\ref{def:top-space}) and the definition of prebase we get the following.

\begin{thm}{Observation}\label{ex:prebase}
Let $\s{P}$ be a prebase for the topology on $\c{Y}$.
Then a map $f\: \c{X} \to \c{Y}$ is continuous if and only if $f^{-1}(P)$ is open for any set $P$ in $\s{P}$.

\end{thm}

There are almost no restrictions on a prebase --- we may start with any family $\s{P}$ of subsets that covers the whole space $\c{X}$ and define a topology by declaring that \textit{$\s{P}$ is a prebase for the topology}.
It defines the weakest topology on $\c{X}$ such that every set of $\s{P}$ is open.

\begin{thm}{Exercise}\label{ex:graph}
Let $\c{X}$ and $\c{Y}$ be topological spaces.

\begin{subthm}{ex:graph:prebase}
Show that the product topology on $\c{X}\times\c{Y}$ can be redefined as a topology with a prebase formed by all products $\c{X}\times W$ and $V\times \c{Y}$, where $V$ is open in $\c{X}$ and $W$ is open in $\c{Y}$.
\end{subthm}

\begin{subthm}{ex:graph:graph}
Apply part \ref{SHORT.ex:graph:prebase} and \ref{ex:prebase} to prove the following:
given a map $f\:\c{X}\z\to \c{Y}$, consider the map $F\:\c{X}\z\to \c{X}\times\c{Y}$ defined by $F\:x\z\mapsto (x,f(x))$.
Show that $f$ is continuous if and only if $F$ is an embedding.
\end{subthm}

\end{thm}

\section{Initial topology}\label{induced-family}

Let $\c{X}$ and $\c{Y}$ be topological spaces.
Note that the product topology on $\c{X}\times\c{Y}$ is the weakest topology such that projections $(x,y)\mapsto x$ and $(x,y)\mapsto y$ are continuous.

Indeed, these projections are continuous if the inverse images of all open sets in $\c{X}$ and $\c{Y}$ are open in $\c{X}\times\c{Y}$.
In other words, the topology on $\c{X}\times\c{Y}$ must contain all sets of the form $V\times \c{Y}$ and $\c{X}\times W$ for open sets $V\subset \c{X}$ and $W\subset \c{Y}$.
By \ref{ex:graph:prebase}, these sets form a prebase in $\c{X}\times\c{Y}$ and its topology is the weakest topology  that contains these sets.

More generally, given a family of maps $f_\alpha\:S\to \c{Y}_\alpha$ from a set $S$ to topological spaces $\c{Y}_\alpha$, we can introduce a topology on $S$ by stating that the inverse images $f^{-1}_\alpha(W_\alpha)$ for open sets $W_\alpha\subset\c{Y}_\alpha$ form its prebase.
It defines the topology on $S$ \index{induced topology}\emph{induced} by the maps $f_\alpha$;
it is the weakest topology on $S$ that makes all maps $f_\alpha$ continuous.
For example, if we apply this construction to the inclusion map $S\hookrightarrow \c{Y}$ of a subset $S\subset \c{Y}$, then we get {}\emph{induced topology} on $S$ (see \ref{sec:induced-topology}).

This construction produces a topology on the source space;
for that reason it is also called the \index{initial topology}\emph{initial topology}.
It can be used to define a topology on an infinite product of spaces, as the induced topology for all its projections.
This topology is called the \index{Tychonoff topology}\emph{Tychonoff topology}.

\begin{thm}{Advanced exercise}\label{ex:induced-nonmetrizable}
Let $\s{O}$ be the topology on $\RR$ induced by the maps $x\mapsto (\cos(t\cdot x),\sin(t\cdot x))$ for all $t\in \RR$.
Show that the space $(\RR,\s{O})$ is not metrizable.
\end{thm}

\section{Final topology}\label{sec:final-topology}

In the previous section, we defined a natural way to move a topology from the target space to the source of a map.
Namely, if $f\:\c{X}\z\to\c{Y}$ is a map, and $\c{Y}$ comes with a topology, then a subset $V\subset\c{X}$ is declared to be open if there is an open subset $W\subset\c{Y}$ such that $V=f^{-1}(W)$.

The following exercise describes an analogous construction that moves a topology from source to target.
It can be solved by checking the conditions in \ref{def:top-space} as we did in \ref{sec:induced-topology}.

\begin{thm}{Exercise}\label{ex:move-topology:pushforward}
Let $f\:\c{X}\to\c{Y}$ be a map between two sets.
Assume $\c{X}$ is equipped with a topology.
Declare a subset $W\subset\c{Y}$ to be open if the subset $V=f^{-1}(W)$ is open in $\c{X}$.
Show that it defines a topology on $\c{Y}$.

\end{thm}

The constructed topology on $\c{Y}$ is called the \index{final topology}\emph{final} topology for $f$.
Note that points in $\c{Y}$ outside the image of $\c{X}$ form open one-point sets;
indeed, if $y\notin f(\c{X})$, then $f^{-1}\{y\}=\emptyset$, and so $\{y\}$ is open.

\begin{thm}{Exercise}\label{ex:pullback/pushforward}
Let $f\:\c{X}\to\c{Y}$ be a continuous map between topological spaces.

\begin{subthm}{}
Show that the initial topology for $f$ on $\c{X}$ is weaker than its own topology.
\end{subthm}

\begin{subthm}{}
Show that the final topology for $f$ on $\c{Y}$ is stronger than its own topology.
\end{subthm}

\end{thm}

\begin{thm}{Exercise}\label{ex:category-def-pull-push}
Let $g\:\c{X}\to\c{Y}$ be a continuous map.

\begin{subthm}{}
Suppose $\c{X}$ is equipped with the initial topology.
Show that a map $f\:\c{W}\to\c{X}$ is continuous if and only if the composition $g\circ f\:\c{W}\to\c{Y}$ is continuous.
\end{subthm}

\begin{subthm}{}
Suppose $\c{Y}$ is equipped with the final topology.
Show that a map $h\:\c{Y}\to\c{Z}$ is continuous if and only if the composition $h\circ g\:\c{X}\z\to\c{Z}$ is continuous.
\end{subthm}

\end{thm}

\begin{thm}{Exercise}\label{ex:open-closed-pushforward}
Let $f\:\c{X}\to \c{Y}$ be a continuous surjective map.
Assume $f$ is (a) open or (b) closed.
Show that $\c{Y}$ has the final topology induced by $f$.
\end{thm}

\section{Quotient topology}

The initial topology is used mostly for injective maps; in this case, it is essentially the same as the \index{induced topology}\emph{induced topology}.
Similarly, the final topology is mostly used for surjective maps.
This particular case of the construction is called the \index{quotient topology}\emph{quotient topology},
which we are about to describe.

Let $\sim$ be an \index{equivalence relation}\emph{equivalence relation} on a topological space $\c{X}$;
that is, for any $x,y,z\in\c{X}$ the following conditions hold:
\begin{itemize}
 \item $x\sim x$;
 \item if $x\sim y$, then $y\sim x$;
 \item if $x\sim y$ and $y\sim z$, then $x\sim z$.
\end{itemize}

Recall that the set
\[[x]=\set{y\in \c{X}}{y\sim x}\]
is called the \index{equivalence class}\emph{equivalence class} of $x$.
The set of all equivalence classes in $\c{X}$ will be denoted by $\c{X}/{\sim}$.

Observe that $x\mapsto [x]$ defines a surjective map $\c{X}\to \c{X}/{\sim}$.
The corresponding final topology on $\c{X}/{\sim}$ is called the \index{quotient topology}\emph{quotient topology}.
By default, $\c{X}/{\sim}$ is equipped with the quotient topology;
in this case, it is called a \index{quotient space}\emph{quotient space}.

The following exercise ties equivalence relations with maps.

\begin{thm}{Exercise}\label{ex:eq-relation-f}
Show that an arbitrary map $f\:\c{X}\to \c{Y}$ defines the following equivalence relation on $\c{X}$:
\[x\sim x'\quad\Leftrightarrow\quad f(x)=f(x').\]
Moreover,
\[f(x)=y\quad\Leftrightarrow\quad [x]=f^{-1}\{y\}.\]
\end{thm}

Given a set $A$ in a topological space $\c{X}$, the space $\c{X}/A$ is defined as the quotient space $\c{X}/{\sim}$ for the minimal equivalence relation such that $a\sim b$ for any $a,b\in A$.

\begin{thm}{Exercise}\label{ex:[0,1]/(0,1)}
Describe the quotient space $[0,1]/(0,1)$, where $[0,1]$ and $(0,1)$ are real intervals;
that is, list the points and the open sets of the quotient space.
\end{thm}

One may also check by hand that the quotient space $[0,1]/\{0,1\}\z\simeq\SS^1$;
that is, the unit segment with identified ends is homeomorphic to the circle.
Theorem \ref{obs:compact-to-hausdorff} will provide a very general tool that helps to prove such statements.

\section{Actions}

Let $\c{X}$ be a set.
A subset $G$ of bijections $\c{X}\to \c{X}$ that includes the identity map $\id_{\c{X}}$ and is closed under composition and inversion is called a \index{group of symmetries}\emph{group of symmetries} of $\c{X}$.
In this case we may say that $G$ \index{action}\emph{acts} on $\c{X}$, briefly $G\acts\c{X}$.

Typically we are interested in group actions that preserve some structure of the set $\c{X}$.
If $\c{X}$ is a topological space, then it is natural to assume that each $g\in G$ is a continuous bijection $\c{X}\to \c{X}$.
In this case the action $G\acts\c{X}$ is called \index{continuous action}\emph{continuous}.
For such an action each $g\in G$ defines a homeomorphism $\c{X}\to \c{X}$.
Indeed, if $g\in G$, then $g^{-1}\in G$; therefore $g$ as well as its inverse must be continuous.
Actions on topological spaces will always be assumed to be continuous.

Given an action $G\acts\c{X}$, we can equip the set $G$ with a product defined as composition;
that is, if $g,h\in G$ then its product $g\cdot h\in G$ is defined as
$(g\cdot h)(x)\df g(h(x))$.
With this operation, $G$ becomes a \index{group}\emph{group}; that is, it meets the following properties:
\begin{itemize}
\item (associativity) for any $f,g,h\in G$ we have
\[(f\cdot g)\cdot h=f\cdot (g\cdot h);\]
\item (existence of neutral element) there exists $1\in G$ such that for every $g\in G$ we have
\[1\cdot g=g\cdot 1=g;\]
\item (inverse) for each $g\in G$ there exists $g^{-1}\in G$ such that
\[g\cdot g^{-1}\z=g^{-1}\cdot g=1.\]
\end{itemize}
Checking these properties is straightforward.
Associativity follows since it holds for composition, and one can take $1=\id_{\c{X}}$.

Finally note that an abstract group (that is, a set $G$ with product that meets the three properties above) acts on itself by $g\: h\mapsto g\cdot h$, so any group acts on some set.
Furthermore, we may equip $G$ with discrete topology and this way we obtain a continuous action $G\acts G$.
So you may think that actions are more fundamental than groups.

For an action $G\acts \c{X}$, it is common to denote $g(x)$ by $g\cdot x$, where $g\in G$ and $x\in\c{X}$.
Note that associativity implies that the expression
\[g\cdot h\cdot x\]
makes sense for any $g,h\in G$ and $x\in \c{X}$; that is, $g\cdot (h\cdot x)=(g\cdot h)\cdot x$.

\section{Orbit spaces}

Choose a continuous action $G\acts \c{X}$.
The set
\[G\cdot x:=\set{g\cdot x\in \c{X}}{g\in G}\]
is called the \index{orbit}\emph{$G$-orbit} of $x$ (or, briefly, {}\emph{orbit}).

Set $x\sim y$ if there is $g\in G$ such that $y=g\cdot x$.
Observe that $\sim$ is an equivalence relation on $\c{X}$.
Indeed,
$x\sim x$ since $x=1\cdot x$.
Further, if $y=g\cdot x$, then
\[x=1\cdot x=g^{-1}\cdot g\cdot  x=g^{-1}\cdot y;\] since $g^{-1}\in G$ we get that $x\sim y$
$\Rightarrow$
$y\sim x$.
Finally, suppose $x\sim y$ and $y\sim z$;
that is, $y=g\cdot x$ and $z=h\cdot y$ for some $g,h\in G$, then $z=(h\cdot g)\cdot x$;
therefore $x\sim z$.

For the described equivalence relation, the quotient space $\c{X}/{\sim}$ can also be denoted by $\c{X}/G$;
it is called the quotient of $\c{X}$ by the action of~$G$.

Note that $[x]=G\cdot x$; that is, the orbit of $x$ coincides with its equivalence class of $\sim$.
For that reason, $\c{X}/G$ is also called the \index{orbit space}\emph{orbit space}.

\begin{thm}{Exercise}\label{ex:R/R}
Positive real numbers $\RR_+$ act on $\RR$ by multiplication.
Describe the quotient space $\RR/\RR_+$;
that is, list the points and the open sets of the quotient space.

\end{thm}

\begin{thm}{Exercise}\label{ex:quotient-map}
Suppose a group $G$ acts on a topological space $\c{X}$ and $f\:\c{X}\to \c{X}/G$ is the quotient map.
\begin{subthm}{ex:quotient-map:open}
Show that $f$ is open.
\end{subthm}
\begin{subthm}{ex:quotient-map:closed}
Assume $G$ is finite.
Show that $f$ is closed.
\end{subthm}

\end{thm}
