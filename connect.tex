\chapter{Connected spaces}\label{chap:connect}

\section{Definitions}

A subset of a topological space is called \index{clopen subset}\emph{clopen} if it is closed and open at the same time.

\begin{thm}{Definition}
A topological space $\c{X}$ is called \index{connected space}\emph{connected} if it has exactly two clopen sets $\emptyset$ and $\c{X}$.

A subset of a topological space is called \index{connected set}\emph{connected} if the corresponding subspace%
\footnote{Which is the subset equipped with the induced topology.}
is connected.
\end{thm}

According to our definition, \textit{the empty space is not connected}.
Not everyone agrees with this convention.%
\footnote{This convention is similar in spirit to saying that $1$ is not prime --- if prime meant \textit{no nontrivial divisors}, then $1$ would be prime, \textit{but it is not}.
Similarly, if a connected space meant \textit{no proper clopen sets}, then the empty set would be connected, but \textit{it is not} (at least in this course).
}

Suppose $V$ is a \index{proper subset}\emph{proper} clopen subset of a topological space $\c{X}$;
that is, $V\ne\emptyset$ and $V\ne\c{X}$.
Note that its complement $W=\c{X}\setminus V$ is also a proper clopen subset.
In particular, there are two open sets $V,W\subset \c{X}$ such that
\[V\ne\emptyset,
\quad
W\ne\emptyset,
\quad
V\cup W=\c{X},
\quad\text{and}\quad
V\cap W=\emptyset.\]
In this case, we say that $V$ and $W$ \index{separating pair}\emph{separate} $\c{X}$.

Spelling out the notion of subspace leads to the following definition:
an open {}\emph{separating pair} of a set $A\subset\c{X}$ consists of two open sets $V,W\subset \c{X}$ such that
\[V\cap A\ne\emptyset,\quad
W\cap A\ne\emptyset,\quad
V\cup W\supset A,\quad\text{and}\quad
V\cap W\cap A=\emptyset.\]

\begin{thm}{Observation}\label{def:connected-subset}
A nonempty topological space (or a nonempty subset of a topological space) is \index{disconnected subset}\emph{disconnected} if and only if it has an open separating pair.
\end{thm}

\begin{thm}{Exercise}\label{ex:connected->R}
Let $f\:\c{X}\to\RR$ be a continuous function defined on a connected space.
Suppose that $f(x_1)>0$ and $f(x_2)<0$ for some $x_1,x_2\in \c{X}$.
Show that there is $x_3\in \c{X}$ such that $f(x_3)=0$.
\end{thm}

\begin{thm}{Observation}\label{obs:A<B<bar-A}
Let $A$ be a connected set in a topological space and
\[A\subset B\subset \bar A.\]
Then $B$ is connected.
\end{thm}

\parit{Proof.}
Since $A$ is connected, we have $A\ne\emptyset$ and hence $B\ne\emptyset$.

Assume $V$ and $W$ form an open separating pair of $B$.
Since $B\subset \bar A$,
\begin{align*}
V\cap B\ne \emptyset \ &\Rightarrow V\ \cap A\ne \emptyset,
\intertext{and}
W\cap B\ne \emptyset \ &\Rightarrow W\ \cap A\ne \emptyset.
\end{align*}
It follows that $V$ and $W$ form an open separating pair of $A$ --- a contradiction.
\qeds

\begin{thm}{Advanced exercise}\label{ex:nested-in-compact}
Let $Q_1\supset Q_2\supset\ldots $ be a nested sequence of closed connected sets in a compact space.
Show that the intersection $Q_\infty=Q_1\cap Q_2\cap\ldots $ is connected.
\end{thm}

\begin{thm}{Advanced exercise}\label{ex:colomb}
Consider the set of positive integers $\NN\z=\{1,2,\dots\}$.
Denote by $\c{B}$ the set of all arithmetic progressions $\{a$, $a+d$, $a+2\cdot d,\dots\}$ in $\NN$ for all relatively prime positive integers $a$ and $d$ such that $a\le d$.

\begin{subthm}{ex:colomb:base}
Show that $\c{B}$ is a base of some topology on $\NN$.
\end{subthm}

\begin{subthm}{ex:colomb:connected}
Show that $\NN$ equipped with this topology is connected, Hausdorff, and noncompact.
\end{subthm}

\end{thm}

\section{Key statements}

\begin{thm}{Proposition}\label{prop:image-connected}
Let $f\:\c{X}\to \c{Y}$ be a continuous map between topological spaces.
Suppose $A\subset \c{X}$ is a connected set.
Then the image $B=f(A)$ is a connected set in $\c{Y}$.
\end{thm}

\parit{Proof.}
Since $A$ is connected, $A\ne \emptyset$, and hence $B\ne\emptyset$.

Assume that $B=f(A)$ is disconnected.
Choose open sets $V$ and $W$ that separate $B$; that is,
\[V\cap B\ne\emptyset,\ W\cap B\ne\emptyset,\ V\cup W\supset B,\ \text{and}\ V\cap W\cap B=\emptyset.
\eqlbl{eq:VWB}\]

Since $f$ is continuous, $V'=f^{-1}(V)$ and $W'=f^{-1}(W)$ are open sets in $\c{X}$.
Note that \ref{eq:VWB} implies that $V'$ and $W'$ separate $A$; that is,
\[
V'\cap A\ne\emptyset,\ W'\cap A\ne\emptyset,\ V'\cup W'\supset A,\ \text{and}\ V'\cap W'\cap A=\emptyset.\]
Therefore, $A$ is disconnected --- a contradiction.
\qeds

\begin{thm}{Exercise}\label{ex:quotient-connected}
Let $\c{X}$ be a connected space.
Show that the quotient space $\c{X}/{\sim}$ is connected for any equivalence relation $\sim$ on $\c{X}$.
\end{thm}

\begin{thm}{Proposition}\label{prop:union-connect}
Assume $\{A_\alpha\}_{\alpha\in\c{I}}$ is a family of connected subsets of a topological space.
If $\bigcap_\alpha A_\alpha\ne\emptyset$,
then
\[A=\bigcup_\alpha A_\alpha\] 
is connected.
\end{thm}

\parit{Proof.}
Assume that $A$ is disconnected;
let $V$ and $W$ be an open separating pair.

Choose $p\in \bigcap_\alpha A_\alpha$.
Without loss of generality, we can assume that $p\in V$.

In particular, $V\cap A_\alpha\ne\emptyset$ for any $\alpha$.
Since $A_\alpha$ is connected, we have that $W\cap A_\alpha=\emptyset$ for each $\alpha$;
otherwise $V$ and $W$ would separate $A_\alpha$.
Therefore 
\begin{align*}
W\cap A&=W\cap\biggl(\bigcup_\alpha A_\alpha\biggr)=
\\&=\bigcup_\alpha(W\cap A_\alpha)=
\\&=\emptyset.
\end{align*}
Hence $V$ and $W$ do not separate $A$ --- a contradiction.
\qeds

\begin{thm}{Exercise}\label{ex:connected-product}
Show that two topological spaces $\c{X}$ and $\c{Y}$ are connected
if and only if the product space $\c{X}\times\c{Y}$ is connected.
\end{thm}


\section{Real interval}

\begin{thm}{Proposition}\label{prop:connected[0,1]}
The real interval $[0,1]$ is connected.
\end{thm}

The proof reuses the construction in \ref{sec:compact-interval}.

\parit{Proof.}
Assume the contrary;
choose an open separating pair of $[0,1]$, say $V$ and $W$.
Fix $a_0\in V$ and $b_0\in W$;
without loss of generality, we can assume that $a_0<b_0$.

Let us construct a nested sequence of closed intervals 
\[[a_0,b_0]\supset [a_1,b_1]\supset \dots\]
such that 
\[b_n-a_n=\frac{b_0-a_0}{2^n},\quad a_n\in V,\quad\text{and}\quad b_n\in W
\eqlbl{eq:ab-in-VW}\]
for any $n$.

The construction is recursive.
Assume $[a_{n-1},b_{n-1}]$ is already constructed.
Let $c=\tfrac12\cdot(a_{n-1}+b_{n-1})$; note that $c\in V$ or $c\in W$.
If $c\in V$,
then set $a_{n}=c$ and $b_{n}=b_{n-1}$;
if $c\in W$, then set $a_{n}=a_{n-1}$ and $b_{n}=c$.
In both cases, the conditions in \ref{eq:ab-in-VW} hold.

Observe that
\[
a_0\le a_1\le \dots \le b_1\le b_0.
\]
In particular, the sequence $a_n$ is nondecreasing and bounded above by~$b_0$.
Therefore, it must converge; denote its limit by $x$.
Since
\[b_n-a_n\z=\tfrac1{2^n}\cdot(b_0-a_0),\] the sequence $b_n$ also converges to $x$.
The point $x$ has to belong to $V$ or~$W$.
Since both $V$ and $W$ are open,
whichever set contains $x$ contains all $a_n$ and $b_n$ for all large $n$, which contradicts \ref{eq:ab-in-VW}.
\qeds

\begin{thm}{Exercise}\label{ex:R-connected}
Show that the real line $\RR$ is a connected space.
Conclude that any continuous map $\RR\to \ZZ$ is constant;
here $\ZZ\subset \RR$ denotes the set of integers.
\end{thm}

Recall that a set $V\subset\RR^n$ is called \index{star-shaped set}\emph{star-shaped} if there exists a point $p\in V$ such that for all $x\in V$ the line segment from $p$ to $x$ lies in~$V$.

\begin{thm}{Exercise}\label{ex:connected-star-shaped}
Show that any star-shaped set in $\RR^n$ is connected.
\end{thm}


\section{Connected components}

Let $x$ be a point in a topological space $\c{X}$.
The union of all connected sets containing $x$ is called the \index{connected component}\emph{connected component} of $x$.
Note that the space $\c{X}$ is connected if and only if a connected component of some (and therefore of any) point is the whole space $\c{X}$.

Observe that two connected components either coincide or are disjoint.
In other words, lying in the same connected component defines an equivalence relation.

\begin{thm}{Exercise}\label{ex:nonopen-connected-component}

\begin{subthm}{ex:nonopen-connected-component:closed}
Show that any connected component is a closed set.
\end{subthm}

\begin{subthm}{ex:nonopen-connected-component:open}
Construct an example of a topological space $\c{X}$ and a point $x\in \c{X}$ such that the connected component of $x$ is not open.
\end{subthm}

\begin{subthm}{ex:nonopen-connected-component:finite}
Suppose that a space $\c{X}$ has a finite number of connected components.
Show that each connected component of $\c{X}$ is open.
\end{subthm}

\end{thm}

\begin{thm}{Advanced exercise}\label{ex:quasi-component}
Let $\c{X}$ be a compact Hausdorff space.
Show that the connected component of a point $x\in \c{X}$ is the intersection of all clopen sets containing $x$.
Find examples showing that this is no longer true without assuming compactness or Hausdorffness.
\end{thm}

\section{Locally connected spaces}\label{sec:Locally connected spaces}

A topological space $\c{X}$ is called \index{locally connected}\emph{locally connected} if for any point $p\in \c{X}$, any neighborhood of $p$ contains a connected neighborhood;
that is, given an open set $W\ni p$ there is a connected open set $V$ such that $p\in V\subset W$.
By \ref{ex:connected-star-shaped}, $\RR^n$ is locally connected for any $n$.

\begin{thm}{Observation}\label{obs:open-connected-component}
Let $V$ be an open set in a locally connected space.
Then any connected component of $V$ is open.
\end{thm}

\parit{Proof.}
Let $V_0$ be a connected component in $V$.
Given a point $x\in V_0$, choose a connected neighborhood $N_x\subset V$ of $x$.
By \ref{prop:union-connect}, $N_x\subset V_0$, and the observation follows.
\qeds

\section{Cut points}

Connectedness is defined in purely topological terms.
In particular, the number of connected components is a topological invariant;
that is, \textit{if two spaces are homeomorphic, then they have the same number of connected components}.
In particular, a connected space is not homeomorphic to a disconnected space.

Let us describe a refined way to apply this observation.
Suppose $\c{X}$ is a connected space.
A point $x\in\c{X}$ is called a \index{cut point}\emph{cut point} of $\c{X}$ if removing $x$ from $\c{X}$ produces a disconnected space;
that is, the subset $\c{X}\setminus\{x\}$ is disconnected.

Note that if $f\:\c{X}\to \c{Y}$ is a homeomorphism, then a point $x \in\c{X}$ is a cut point of $\c{X}$ if and only if $y=f(x)$ is a cut point of $\c{Y}$.
Moreover, the number of connected components in $\c{X}\setminus\{x\}$ is the same as in $\c{Y}\setminus\{y\}$.
The latter follows since the restriction of $f$ defines a homeomorphism $\c{X}\setminus\{x\}\to \c{Y}\setminus\{y\}$.

For example, consider the circle $\mathbb{S}^1$ and the line segment $[0,1]$.
Note that both spaces are connected, compact, Hausdorff, metrizable, and Baire --- none of these properties can distinguish these spaces.
Now choose the midpoint $\tfrac12\in [0,1]$;
observe that its complement $[0,1]\setminus \{\tfrac12\}$ has two connected components.
If there is a homeomorphism $f\:[0,1]\to \mathbb{S}^1$, then the set $\mathbb{S}^1\setminus \{f(\tfrac12)\}$ must have two connected components.
On the other hand, $\mathbb{S}^1\setminus \{x\}$ is connected for any point $x\in \mathbb{S}^1$ (prove it).
This leads to a contradiction, proving that $\mathbb{S}^1$ is not homeomorphic to the line segment $[0,1]$.

The same type of argument can be used to solve the following exercises.
