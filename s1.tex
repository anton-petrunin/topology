\chapter{Fundamental group of a circle}

In this chapter we will caclulate fundamental group of a circle;
it will give a first example of a space with a nontrivial fundamental group.
In the proof one has to show nonexistence of homotopies between certain loops;
this is part requres new ideas.

\section{The statement}

\begin{thm}{Theorem}\label{thm:pi1(S1)}
The fundamental group of the circle $\mathbb{S}^1$ is isomorphic to the additive group of integers $\mathbb{Z}$.
\end{thm}

The proof is done in several steps.
Given a loop $f\:[0,1]\to \mathbb{S}^1$, denote by $\deg f$ the number of times it goes aroond $\mathbb{S}^1$ counteclockwise (each clockwise turn is counded with minus).
In the following sections we will define degree formally, prove that it depends only on the homotopy class of the loop and then show that $\deg\:\pi_1(\SS^1)\to\ZZ$ is an isomorphism.

\begin{thm}{Exercise}\label{ex:S1}
Apply the theorem to show the following.

\begin{subthm}{ex:S1:R3>R^2}
The space $\RR^3$ is not homeomorphic to the plane $\RR^2$.
\end{subthm}

\begin{subthm}{ex:D-->S}
The circle $\SS^1=\partial \DD^2$ is not a retract of the disc $\DD^2$.
\end{subthm}

\begin{subthm}{ex:bry->bry}
Given $x\in\DD^2$ the complement $\DD^2\setminus\{x\}$ is simply connected if and only if $x\in \SS^1=\partial \DD^2$.
Conclude that any homeomorphism $\DD^2\to\DD^2$ maps $\partial \DD$ to itself.
\end{subthm}

\begin{subthm}{ex:S1:finite-complement}
The complement of any finite nonempty set $F\subset\RR^2$ is not contractible.
\end{subthm}

\end{thm}

Part \ref{SHORT.ex:D-->S} in the last exercise has the following application.

\begin{thm}{Brouwer fixed point theorem in the plane}
\label{cor:brouwer-D2}
Any continuous map $f\:\DD^2\to\DD^2$ has a fixed point.
\end{thm}

\section{Lifts}\label{sec:lifting-s1}

\begin{wrapfigure}{r}{16 mm}
\vskip-14mm
\centering
\includegraphics{mppics/pic-1525}
\end{wrapfigure}

Let us identify $\RR^2$ with the complex plane $\CC$;
namely, we will encode a point $(x,y)\in\RR^2$ by the complex number $z=x+i\cdot y$; here $i$ denotes the imaginary unit.
In particular,
\[\SS^1=\set{z=x+i\cdot y\in \CC}{|z|^2=x^2+y^2=1}.\]

Consider the map $w\:\RR\to \mathbb{S}^1\subset \CC$ defined by
\[w\:x\mapsto \exp(2\cdot\pi\cdot i\cdot x)=\cos(2\cdot\pi\cdot x)+i\cdot\sin(2\cdot\pi\cdot x)).\]
(The last equality is called Euler's idenity, it can be taken as a definition of $\exp(2\cdot\pi\cdot i\cdot x)$.)

Let $f\:\c{X}\to \SS^1$ be a continuous map.
A continuous map $\tilde f\:\c{X}\to \RR$ will be called \index{lift}\emph{lift} of $f$ if $f=w\circ \tilde f$;
in other words if the following diagram \index{commutative diagram}\emph{commutes}\footnote{It means that following directed paths in the diagram with the same start and endpoints lead to the same result.
Even in this simple case, using a commutative diagram is helpful, and later on you simply can't do without them.}.
\begin{figure}[h!]
\centering
\begin{tikzcd}
& \RR \arrow[d, "w"]\\
\c{X} \arrow[r, "f"']\arrow[ru, "\tilde f"] & \SS^1
\end{tikzcd}
\end{figure}


\begin{thm}{Proposition}\label{prop:lifting}
Let $f_0\:[0,1]\to\SS^1$ be a path and let $x_0\in \RR$ be a point such that $w(x_0)=f_0(0)$.
Then there is unique lift $\tilde f_0$ of $f$ such that $\tilde f_0(0)=x_0$.

Moreover, given a homotopy of paths $f_\tau$ from $f_0$ to $f_1$ there is unique lift $\tilde f_\tau$ such that $\tilde f_0(0)=x_0$.
In particular, if $f_0\sim f_1$ then $\tilde f_0\sim \tilde f_1$.
\end{thm}

\begin{thm}{Lemma}\label{lem:covering}
Let $W=\SS^1\setminus \{u_0\}$ for some $u_0=w(x_0)\in\SS^1$.
Then $w^{-1}(W)$ is a disjoint union of open intervals $V_n=(x_0+n,x_0+n+1)$ for $n\in \ZZ$
and $w$ defines a homeomorphism $V_n\to U$ for each $n$.

In particular, given a connected space $\c{X}$, continuous map $f\:\c{X}\z\to W$, and a point $q\in \RR$ such that $w(q)=f(p)$ for some $p\in \c{X}$ there is a unique lift $\tilde f\:\c{X}\to \RR$ of $f$ such that $\tilde f(p)=q$.
\end{thm}


\parit{Proof.}
Observe that $w_n=w|_{V_n}$ is homeomorphism $V_n\to W$.
Indeed, $w_n$ is a continuous bijection.
Furthermore, by \ref{obs:compact-to-hausdorff} $w|_{\bar V_n}$ is a closed map.
Therefore $w_n=w|_{V_n}$ is a closed continuous bijection $V_n\to W$ and hence a homeomorphism.

Note that the intervals $V_n$ are connected components of $w^{-1}W$.
In particular, $q\in V_n$ for some $n$.
Since $\c{X}$ is connected, $\tilde f(\c{X})\subset V_n$ for any lift $\tilde f$ of $f$ such that $w(q)=f(p)$.
Since $w_n=w|_{V_n}$ is a homeomorpism, we have $\tilde f=w_n^{-1}\circ f$, which proves existance and uiqueness.
\qeds


\parit{Proof of \ref{prop:lifting}.}
Consider two open subsets $W_1=\SS^1\setminus \{1\}$ and $W_2\z=\SS^1\setminus \{-1\}$.
Note that $\SS^1=W_1\cup W_2$.
Therefore,
\[[0,1]=f^{-1}(W_1)\z\cup f^{-1}(W_2).\]

Let $\eps>0$ be the Lebesgue number of this covering; see \ref{prop:lebesgue-number}.
Consider a partition $0=t_0<t_1<\dots<t_n=1$ of $[0,1]$ into equal intervals shorter than $\eps$.
Note that $f([t_i,t_{i+1}])\subset W_1$ or  $f([t_i,t_{i+1}])\subset W_2$ for each $i$.

By \ref{lem:covering}, there is unique lift of $f|_{[t_0,t_1]}$ such that $\tilde f(0)=x_0$; let $x_1=\tilde f(t_1)$.
By the same argument, there is unique lift of $f|_{[t_1,t_2]}$ such that $\tilde f(t_1)=x_1$.
Repeating this argument finitely many times produces a lift of $f$, which is unique by construction.
The continuity of $f$ follows from \ref{ex:u-continuous:closed}.

The proof of the last statement is similar.
Choose a homotopy $F\:[0,1]\times [0,1]\to\SS^1$ from $f_0$ to $f_1$.
Applying \ref{prop:lebesgue-number} again, we can subdivide $[0,1]\times [0,1]$ into small squares by horizontal and vertical lines so that $F$ maps each small square in $W_1$ or in $W_2$.
Then we can build the lift $\tilde F$ square by square, from button up in the order shown in the picture.

When we extending $\tilde F$ over a small square $\square$ the value of $\tilde F$ is already defined at its lower left corner, say $v$.
So we can apply \ref{lem:covering}.
The part of $\square$ on which $\tilde F$ was already defined is one or two sides adjacent to $v$.
In both cases this set is connected and by \ref{lem:covering} the extension agrees with the already constructed part.
Again, the continuity of $\tilde F$ follows from \ref{ex:u-continuous:closed}.\qeds

\section{Degree}

A loop in $\SS^1$ based at $1$ is a continuous map $f\:[0,1]\to\SS^1$
such that $f(0)=f(1)=1$.
Let $\tilde f\:[0,1]\to\RR$ be the unique lift with $\tilde f(0)=0$ provided by \ref{prop:lifting}.
Since $w^{-1}(1)=\ZZ$, we have $\tilde f(1)\in\ZZ$.
The integer $\tilde f(1)$ will be called \index{degree of loop}\emph{degree} of the loop $f$, it will be denoted by $\deg f$.


\begin{thm}{Theorem}\label{thm:pi1S1Z}
The map $[f]\mapsto \deg f$ defines an isomorphism
\[\pi_1(\SS^1,1)\to\ZZ.\]
\end{thm}

\parit{Proof.}
Suppose $f_0$ and $f_1$ are two homotopic loops in $\SS^1$ with base point $1$.
Let $f_\tau$ be the corresponding homotopy and $\tilde f_\tau$ be its lift to $\RR$.
By \ref{prop:lifting}, it has a unique lift $\tilde f_\tau\:[0,1]\to \RR$ such that $f_0(0)=0$.

Observe that the function $\tau\mapsto =f_\tau(1)=\deg f_\tau$ is continuous integer valued function defined on $[0,1]$.
Hence this function is constant; in particular $\deg f_0=\deg f_1$

It follows that $\deg f$ depends only on the homotopy class of $f$.
In other words the map $[f]\mapsto \deg f$ is well defined.

Choose two paths $\tilde f_0$ and $\tilde f_1$ based at $1$ in $\SS^1$.
Let $\tilde f_0$ and $\tilde f_1$ be their lifts such that $\tilde f_0(0)=\tilde f_1(0)=0$.
Observe that the path $t\mapsto \deg f_0+\tilde f_1(t)$ is a lift of $f_1$ that starts at $\deg f_0=\tilde f_0(1)$.
Concatenating $\tilde f_0$ with this path we get a lift of $f_0*f_1$ with the endpoint at $\deg f_0+\deg f_1$.
Hence $\deg (f_0*f_1)=\deg f_0+\deg f_1$ for any two loops $f_0$ and $f_1$, and we proved that degree defines a homomorphism $\deg\:\pi_1(\SS^1,1)\cong\ZZ$.

Surjectivity of $\deg\:\pi_1(\SS^1,1)\cong\ZZ$ follows follows since given the path $f_n\:t\mapsto w(n\cdot t)$ has degree $n$.

Let $f_0$ and $f_1$ be two loops at $\SS^1$ with base point $1$, and let $\tilde f_0$ and $\tilde f_1$ be their lifts such that $\tilde f_0(0)=\tilde f_1(0)=0$.
If $\deg f_0=\deg f_1$, then $\tilde f_0(1)=\tilde f_1(1)$.
Therefore there is linear homotopy of paths $\tilde f_\tau$ from $\tilde f_0$ to $\tilde f_1$.
Note that $f_\tau=e\circ \tilde f_\tau$ is a homotopy from $f_0$ to $f_1$.
Therefore, if $\deg f_0=\deg f_1$, then $f_0\sim f_1$;
in other words $\deg$ is an injective homomorphism.
\qeds

\section{Fundamental theorem of algebra}

\begin{thm}{Theorem}\label{cor:FTA}
Every non-constant complex polynomial has a complex root.
\end{thm}

\parit{Proof.}
Assume that a polynomial $p$ has no roots.

Recall that $w(t)=\exp(2\cdot\pi\cdot i\cdot t)$.
Given $r\ge 0$, consider the loop
\[f_r\:t\mapsto p(r\cdot w(t))/|p(r\cdot w(t))|\]
in $\SS^1$.
Observe that $f_0$ is a constant loop.
Note that $f_r$ converges to $w(k\cdot t)$ as $r\to\infty$, here $k$ is
the degree of $p$.
In particular, if $r$ is large then $f_r(t)-w(k\cdot t)\ne 0$ for any $t$;
arguing as in \ref{ex:homot-sphere}, we get that $f_r$ is homotopic to path $t\mapsto w(k\cdot t)$ for large $r$.
This forces a change of degree, which is impossible under a homotopy of paths.
\qeds
