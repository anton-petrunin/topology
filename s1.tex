\chapter{Fundamental group of a circle}\label{chap:S1}

In this chapter we will calculate the fundamental group of a circle;
it will give a first example of a space with a nontrivial fundamental group.
In the proof one has to show nonexistence of homotopies between certain loops;
this part requires new ideas.

\section{The statement}

\begin{thm}{Theorem}\label{thm:pi1(S1)}
The fundamental group of the circle $\mathbb{S}^1$ is isomorphic to the additive group of integers $\mathbb{Z}$.
\end{thm}

The proof is done in several steps.
Given a loop $a\:[0,1]\to \mathbb{S}^1$, denote by $\deg a$ the number of times it goes around $\mathbb{S}^1$ counterclockwise (each clockwise turn is counted with minus).
In the following sections we will define degree formally, prove that it depends only on the homotopy class of the loop and then show that $\deg\:\pi_1\, \SS^1 \to\ZZ$ is an isomorphism.

Let us give one application of the theorem before proving it.

\begin{thm}{Brouwer fixed point theorem in the plane}
\label{cor:brouwer-D2}
Any continuous map $f\:\DD^2\to\DD^2$ has a fixed point.
\end{thm}



\parit{Proof.}
Arguing by contradiction, assume $x\z\neq f(x)$ for all $x\in\DD^2$.
Given $x\in\DD^2$ consider the half-line $H$ from $f(x)$ to $x$.%
\footnote{Remember that $H$ goes backwards, from $f(x)$ to $x$, not the other way around.}

\section{Wrapping map}\label{sec:lifting-s1}

\begin{wrapfigure}{r}{16 mm}
\vskip-14mm
\centering
\includegraphics{mppics/pic-1525}
\end{wrapfigure}

Let us identify $\RR^2$ with the complex plane $\CC$;
namely, we will encode a point $(x,y)\in\RR^2$ by the complex number $z=x+i\cdot y$; here $i$ denotes the imaginary unit.
In particular,
\[\SS^1=\set{z=x+i\cdot y\in \CC}{|z|\df\sqrt{x^2+y^2}=1}.\]

Consider the wrapping map $w\:\RR\to \mathbb{S}^1\subset \CC$ defined by
\[w\:x\mapsto \exp(2\cdot\pi\cdot i\cdot x)=\cos(2\cdot\pi\cdot x)+i\cdot\sin(2\cdot\pi\cdot x).\]
(The last equality follows from Euler's formula $\exp(i\cdot x)=\cos x \z+i\cdot \sin x$, which can be taken as a definition of $\exp(i\cdot x)$.)

\begin{wrapfigure}[5]{r}{23 mm}
\vskip-0mm
\centering
\begin{tikzcd}
& \RR \arrow[d, "w"]\\
\c{X} \arrow[r, "a"']\arrow[ru, "\tilde a"] & \SS^1
\end{tikzcd}
\end{wrapfigure}

Let $a\:\c{X}\to \SS^1$ be a continuous map.
A continuous map $\tilde a\:\c{X}\to \RR$ will be called a \index{lift}\emph{lift} of $a$ if $a=w\circ \tilde a$.
In other words if the diagram \index{commutative diagram}\emph{commutes}, which means that following directed paths in the diagram with the same start and endpoints lead to the same result.
%(Even in this simple case, using a commutative diagram is helpful, and later on you simply can't do without them.)



\begin{thm}{Proposition}\label{prop:lifting}
Let $a_0\:[0,1]\to\SS^1$ be a path and let $x_0\in \RR$ be a point such that $w(x_0)=a_0(0)$.
Then there is a unique lift $\tilde a_0$ of $a_0$ such that $\tilde a_0(0)=x_0$.

Moreover, given a homotopy of paths $a_\tau$ from $a_0$ to $a_1$ there is a unique lifted homotopy of paths $\tilde a_\tau$ such that $\tilde a_0(0)=x_0$.
In particular, if $a_0\sim a_1$ then $\tilde a_0(1)= \tilde a_1(1)$.
\end{thm}

\begin{thm}{Lemma}\label{lem:covering}
Let $W=\SS^1\setminus \{u_0\}$ for some $u_0=w(x_0)\in\SS^1$.
Then $w^{-1}(W)$ is a disjoint union of open intervals $V_n=(x_0+n,x_0+n+1)$ for $n\in \ZZ$
and $w$ defines a homeomorphism $V_n\to W$ for each $n$.

In particular, given a connected space $\c{X}$, continuous map $a\:\c{X}\z\to W$, and a point $q\in \RR$ such that $w(q)=a(p)$ for some $p\in \c{X}$ there is a unique lift $\tilde a\:\c{X}\to \RR$ of $a$ such that $\tilde a(p)=q$.
\end{thm}


\parit{Proof.}
Observe that the restriction $w_n=w|_{V_n}$ is homeomorphism $V_n\to W$.
Indeed, $w_n$ is a continuous bijection.
Furthermore, by \ref{obs:compact-to-hausdorff} $w|_{\bar V_n}$ is a closed map.
Therefore, $w_n$ is a continuous bijection $V_n\to W$ that sends closed sets in the induced topology of $V_n$ to closed sets in the induced topology of $W$; hence $w_n$ is a homeomorphism.

Note that the intervals $V_n$ are connected components of $w^{-1}(W)$.
In particular, $q\in V_n$ for some $n$.
Since $\c{X}$ is connected, $\tilde a(\c{X})\z\subset V_n$ for any lift $\tilde a$ of $a$ such that $\tilde a(p)=q$.
Since $w_n=w|_{V_n}$ is a homeomorphism, we have $\tilde a=w_n^{-1}\circ a$, which proves both, existence and uniqueness.
\qeds

\parit{Proof of \ref{prop:lifting}.}
Consider two open subsets $W_1=\SS^1\setminus \{1\}$ and $W_2\z=\SS^1\setminus \{-1\}$.
Note that $\SS^1=W_1\cup W_2$.
Therefore,
\[[0,1]=a_0^{-1}(W_1)\z\cup a_0^{-1}(W_2).\]

The open sets $a_0^{-1}(W_1)$ and $a_0^{-1}(W_2)$ cover $[0,1]$;
let $\eps>0$ be the Lebesgue number (see \ref{prop:lebesgue-number}).
Consider a partition $0=t_0<t_1<\dots<t_n=1$ of $[0,1]$ into equal intervals shorter than $\eps$.
Note that $a_0([t_i,t_{i+1}])\subset W_1$ or  $a_0([t_i,t_{i+1}])\subset W_2$ for each $i$.

By \ref{lem:covering}, we can lift $a_0|_{[t_0,t_1]}$ to $\tilde a_0 \:[t_0,t_1]\to \RR$ such that $\tilde a_0(0)\z=x_0$; let $x_1=\tilde a_0(t_1)$.
By the same argument, we can extend $\tilde a_0$ to $[t_0,t_2]$ via lift of $a_0|_{[t_1,t_2]}$ such that $\tilde a_0(t_1)=x_1$.
The continuity of $\tilde a_0$ follows from \ref{ex:u-continuous:closed}.
Repeating this argument $n$ times gives us a lift of $a_0$, and it is unique by construction.


\begin{wrapfigure}{r}{30 mm}
\vskip-2mm
\centering
\includegraphics{mppics/pic-1410}
\end{wrapfigure}

The proof of the last statement is similar.
Choose a homotopy $H\:[0,1]\times [0,1]\to\SS^1$ from $a_0$ to $a_1$.
Applying \ref{prop:lebesgue-number} again, we can subdivide $[0,1]\times [0,1]$ into small squares by horizontal and vertical lines so that $H$ maps each small square in $W_1$ or in $W_2$.
Then we can build the lift $\tilde H$ square by square, from bottom up in the order shown in the picture.

When we extend $\tilde H$ over a small square $\square$ the value of $\tilde H$ is already defined at its lower left corner, say $v$.
So we can apply \ref{lem:covering}.
The part of $\square$ on which $\tilde H$ was already defined is one or two sides adjacent to $v$.
In both cases this set is connected and by \ref{lem:covering} the extension agrees with the already constructed part.
Again, the continuity of $\tilde H$ follows from \ref{ex:u-continuous:closed}.

It remains to show that for the constructed lift $\tilde a_\tau(t)\df \tilde H(t,\tau)$ the point $\tilde a_\tau(1)$ is constant, so $\tilde a_\tau$ is a true homotopy of paths (not just a free homotopy).
Observe that $w\circ \tilde a_\tau(1)=a_\tau(1)$ which is a constant point.
Therefore the continuous function $\tau\mapsto \tilde a_\tau(1)-\tilde a_0(1)$ takes integer values, and hence it must be a constant function.
Since $0\mapsto 0$, we have $\tau\mapsto 0$ for any $\tau$; hence the last statement follows.
\qeds

\section{Degree}

Recall that we consider $\SS^1$ as a subset of complex plane $\CC$.

A loop in $\SS^1$ based at $1$ is a continuous map $a\:[0,1]\to\SS^1$
such that $a(0)=a(1)=1$.
Let $\tilde a\:[0,1]\to\RR$ be the unique lift with $\tilde a(0)=0$ provided by \ref{prop:lifting}.
Since $w^{-1}(1)=\ZZ$, we have $\tilde a(1)\in\ZZ$.
The integer $\tilde a(1)$ will be called \index{degree of loop}\emph{degree} of the loop $a$, it will be denoted by $\deg a$.


\begin{thm}{Theorem}\label{thm:pi1S1Z}
The map $[a]\mapsto \deg a$ defines an isomorphism
\[\pi_1(\SS^1,1)\to\ZZ.\]
\end{thm}

\parit{Proof.}
Suppose $a_0$ and $a_1$ are two homotopic loops in $\SS^1$ with base point $1$,
and let $\tilde a_0$ and $\tilde a_1$ be their lifts to $\RR$ based at $0$.
By \ref{prop:lifting},
\[\deg a_0=\tilde a_0(1)=\tilde a_1(1)=\deg a_1.\]
It follows that the degree depends only on the homotopy class of a loop.
In other words, $[a]\mapsto \deg a$ is a well-defined map $\pi_1(\SS^1,1)\to \ZZ$.

Choose a pair of loops $a$ and $b$ based at $1$ in $\SS^1$.
Let $\tilde a$ and $\tilde b$ be their lifts such that $\tilde a(0)=\tilde b(0)=0$.
Observe that the path $\tilde b_a\:t\mapsto \deg a+\tilde b(t)$ is a lift of $b$ that starts at $\deg a=\tilde a(1)$.
Therefore $\tilde a*\tilde b_a$ is a lift of $a*b$ that starts at $0$.
Hence
\[\deg (a*b)=(\tilde a*\tilde b_a)(1)=\tilde b_a(1)=\deg a+\deg b\]
for any two loops $a$ and $b$.
Since $[a]\cdot [b]=[a*b]$, we proved that $\deg\:\pi_1(\SS^1,1)\to\ZZ$ is a homomorphism.

Surjectivity of $\deg\:\pi_1(\SS^1,1)\to\ZZ$ follows since the loop $a_n\:t\z\mapsto w(n\cdot t)$ has degree $n$ for any $n\in\ZZ$.

Let $a$ and $b$ be two loops in $\SS^1$ with base point $1$, and let $\tilde a$ and $\tilde b$ be their lifts such that $\tilde a(0)=\tilde b(0)=0$.
If $\deg a=\deg b$, then $\tilde a(1)=\tilde b(1)$.
Therefore there is linear homotopy of paths $\tilde a_\tau$ from $\tilde a$ to $\tilde b$.
Note that $a_\tau=w\circ \tilde a_\tau$ is a homotopy from $a$ to $b$;
therefore, $a\sim b$.
It follows that the homomorphism $\deg\:\pi_1(\SS^1,1)\to\ZZ$ is injective.
\qeds

\section{Fundamental theorem of algebra}

\begin{thm}{Theorem}\label{cor:FTA}
Every non-constant complex polynomial has a complex root.
\end{thm}

\parit{Proof.}
Choose a polynomial
\[p(z)=c_n\cdot z^n+c_{n-1}\cdot z^{n-1}+\dots+ c_0.\]
If $p$ is not constant, we can assume that $n\ge 1$ and $c_n\ne 0$.

Choose $r\ge 0$, and let
\[a_r(t)\df\frac{p(r\cdot w(t))\,\cdot\, |p(r\cdot w(0))|}{|p(r\cdot w(t))|\,\cdot\, p(r\cdot w(0))};
\eqlbl{eq:ar}\]
here $w\:t\mapsto\exp(2\cdot\pi\cdot i\cdot t)$ is the wrapping map.
If $p$ has no roots, then denominator in \ref{eq:ar} does not vanish;
therefore the map $(t,r)\mapsto a_r(t)$ is continuous.
Furthermore, by \ref{eq:ar} we have $a_r(0)=a_r(1)=1$ and $|a_r(t)|\equiv 1$; so $a_r$ is a loop in $\SS^1$ based at $1$.

Observe that $a_0$ is a constant loop and $a_r\sim a_0$ for any $r$.
In particular, $\deg a_r=0$ for any $r$.

On the other hand, consider the loop $a_\infty(t)\df w(n\cdot t)$; note that $\deg a_\infty=n$.

It is straightforward to check that $\frac{p(z)}{z^n}\to c_n$ and $\frac{|p(z)|}{|z^n|}\to |c_n|$ as $|z|\to\infty$.
By \ref{eq:ar}, we have
$|a_r(t)-a_\infty (t)|<1$ for any large~$r$.
Applying linear homotopy with radial projection to $\SS^1$, we get the following homotopy from $a_r$ to $a_\infty$; namely,
\[H(t,s)=\frac{(1-s)\cdot a_r(t)+s\cdot a_\infty(t)}{|(1-s)\cdot a_r(t)+s\cdot a_\infty(t)|}.\]
In particular $a_r\sim a_\infty$ and $\deg a_r=n$ for any large $r$ --- a contradiction.
\qeds
