\chapter{Metric spaces revisited}\label{chap:metric+}

Recall that any metric space comes with a topology.
In particular, we may talk about \textit{compact metric spaces}.
In this chapter we discuss specific properties of metric spaces that are related to compactness.

\section{Lebesgue number}

\begin{thm}{Theorem}\label{prop:lebesgue-number}
Given an open cover $\{V_\alpha\}$ of a compact metric space $\c{M}$,
there is $\eps > 0$ such that every $\eps$-ball in $\c{M}$ lies in $V_\alpha$ for some~$\alpha$.
\end{thm}

The number $\eps$ that meets the condition in the theorem is called the \index{Lebesgue number}\emph{Lebesgue number} of the cover.

\parit{Proof.}
Given a point $p\in \c{M}$ we can choose $r=r(p)>0$ such that the ball $\Ball(p,2\cdot r)$ lies in $V_\alpha$ for some $\alpha$.
Observe that all balls $\Ball(p,r(p))$ form an open cover of $\c{M}$.
Since $\c{M}$ is compact, we can choose a finite subcover $\{\Ball(p_1,r_1),\ldots,\Ball(p_n,r_n)\}$.

Let $\eps=\min\{r_1,\ldots,r_n\}$.
For any $p\in\c{M}$ we can choose a ball $\Ball(p_i,r_i)\ni p$.
Observe that $\Ball(p,\eps)\subset \Ball(p_i,2\cdot r_i)$.
Since $\Ball(p_i,2\cdot r_i)$ lies in some $V_{\alpha}$, so is $\Ball(p,\eps)$.
\qeds


\begin{thm}{Exercise}\label{ex:lebesgue=1}
Construct a noncompact metric space $\c{M}$ such that $1$ is a Lebesgue number for any open cover of $\c{M}$.
\end{thm}

\section[\texorpdfstring{Compactness $\Rightarrow$ sequential compactness}{Compactness ⇒ sequential compactness}]{Compactness $\bm{\Rightarrow}$ sequential compactness}\label{sec:comp=>seq-comp}

A topological space is called \index{sequentially compact space}\emph{sequentially compact} if every sequence in it has a converging subsequence.

{\sloppy

\begin{thm}{Exercise}\label{ex:product-sequentially-compact}
Show that the product of two sequentially compact spaces is sequentially compact.
\end{thm}

}

For general topological spaces sequential compactness does not imply compactness, and the other way around.%
\footnote{An example of compact, but not sequentially compact space has been constructed in \ref{ex:power-power}.
A standard example of sequentially compact, but not compact space is $[0,\Omega)$ --- the set of all countable ordinals, with the order topology;
it is discussed in the classical book of Lynn Arthur Steen and J. Arthur, Jr. Seebach~\cite{steen-seebach}.
}
The following proposition states that these two notions are equivalent for metric spaces.

\begin{thm}{Proposition}\label{prop:seq-comp-metr}
A metric space $\c{M}$ is compact if and only if it is sequentially compact.
\end{thm}

In this section, we prove the only-if part.
The if part will be done in \ref{sec:seq-comp-if} after proving a sequence of auxiliary statements.

\parit{Proof of the only-if part in \ref{prop:seq-comp-metr}.}
Choose a sequence $x_1,x_2,\ldots{}\in \c{M}$.

Note that a point $p\in \c{M}$ is a limit of a subsequence of $x_n$ if and only if for any $\eps>0$, the ball $\Ball(p,\eps)$
contains infinitely many elements~$x_n$.
Indeed, if this property holds, then we can choose $i_1$ such that $x_{i_1}\in \Ball(p,1)$, further $i_2>i_1$ such that $x_{i_2}\in \Ball(p,\tfrac12)$ and so on; on the $n$-th step we get $i_n>i_{n-1}$ such that $x_{i_n}\in \Ball(p,\tfrac1n)$.
The obtained subsequence $x_{i_1},x_{i_2},\ldots$ converges to $p$.


Now assume $\c{M}$ is not sequentially compact;
that is, a sequence $x_1,x_2,\ldots{}\in \c{M}$ has no converging subsequence.
Then for any point $p$ there is $\eps_p>0$ such that $\Ball(p,\eps_p)$ contains only finitely many elements of $x_n$.
Note that all balls $\Ball(p,\eps_p)$ form a cover of~$\c{M}$.
Since the sequence is infinite, this cover does not have a finite subcover,
in particular, $\c{M}$ is not compact.
\qeds

\section{Complete spaces}

A sequence $x_1,x_2,\ldots$ of points in a metric space is called \index{Cauchy sequence}\emph{Cauchy} if
for any $\eps>0$ there is $n$ such that $|x_i-x_j|<\eps$ for all $i,j>n$.
It is easy to prove that any converging sequence is Cauchy, the converse does not hold in general.
A metric space $\c{M}$ is called \index{complete space}\emph{complete} if any Cauchy sequence in $\c{M}$ converges to a point in $\c{M}$.

The Cauchy test (which is assumed to be known from calculus) implies that the real line $\RR$ with the standard metric is complete.
On the other hand, an open interval $(0,1)$ forms a noncomplete subspace of $\RR$;
indeed, the sequence $x_n=\tfrac 1{2^n}$ is Cauchy, but it does not converge in $(0,1)$ --- it converges to $0$ in $\RR$, but $0\notin(0,1)$.


\begin{thm}{Proposition}\label{prop:compact-complete}
Any (sequentially) compact metric space $\c{M}$ is complete.
\end{thm}

\parit{Proof.}
Choose a Cauchy sequence $x_1,x_2,\ldots\in \c{M}$.
By the only-if part of \ref{prop:seq-comp-metr} (which is proved already), we can assume that the space is sequentially compact; therefore, we can find a converging subsequence; denote its limit by $p$.

Choose $\eps>0$ and let $n$ be as in the definition of Cauchy sequence, so $|x_i-x_j|<\eps$ for all $i,j>n$.
Note that the ball $\Ball(p,\eps)$ contains infinitely many elements of the sequence; in particular, $|x_j-p|<\eps$ for some $j>n$.
By the triangle inequality,
\[|x_i-p|\le|x_i-x_j| +|x_j-p|<2\cdot \eps\]
for any $i>n$.
Since $\eps>0$ is arbitrary, the sequence $x_1,x_2,\ldots$ converges to $p$.
\qeds

\begin{thm}{Exercise}\label{ex:closed-complete}

\end{thm}

\begin{thm}{Exercise}\label{ex:compact-complete}
Show that any sequentially compact metric space $\c{M}$ is \index{bounded metric space}\emph{bounded};
that is, $\c{M}$ has finite \index{diameter}\emph{diameter}:
\[\diam\c{M}=\sup\set{|x-y|_{\c{M}}}{x,y\in \c{M}}<\infty.\]
Give an example of noncompact complete bounded metric space.

\end{thm}


\begin{thm}{Exercise}\label{ex:contracction}
Let $\c{M}$ be a complete nonempty metric space, and let
$f\:\c{M}\to \c{M}$ be a {}\emph{contraction map};
that is, there is $\lambda<1$ such that
\[|f(x)-f(y)|_{\c{M}}\le \lambda\cdot|x-y|_{\c{M}}\]
for any $x,y\in \c{M}$.
Show that $f$ has a unique \index{fixed point}\emph{fixed point};
that is, a point $p\in\c{M}$ such that $f(p)=p$.
\end{thm}

\begin{thm}{Exercise}\label{ex:alm-min}
Let $\c{M}$ be a complete metric space, and let $r\:\c{M}\to \RR$ be a positive continuous function.
Show that for any $\eps>0$, there exists a point $p\in \c{M}$ such that
\[
r(x)> (1-\eps)\cdot r(p)
\]
for every $x\in \Ball (p, r(p))$.

\end{thm}


\section{Nets and separability}

Let $\c{M}$ be a metric space.
A subset $A\subset\c{M}$ is called an \index{$\eps$-net}\emph{$\eps$-net} of $\c{M}$ if for any $p\in \c{M}$  there is $a\in A$ such that $|p-a|_{\c{M}}<\eps$.

\begin{thm}{Lemma}\label{lem:net}
Let $\c{M}$ be a sequentially compact metric space.
Then for any $\eps>0$ there is a finite $\eps$-net in $\c{M}$.
\end{thm}

\parit{Proof}.
Choose $\eps>0$.
Consider the following recursive procedure. 

Choose a point $x_1$ in $\c{M}$.
Further, choose a point $x_2$ such that $|x_1\z-x_2|>\eps$.
Further, choose a point $x_3$ such that $|x_1\z-x_3|>\eps$ and $|x_2\z-x_3|>\eps$; and so on.
On the $n$-th step we choose a point $x_n$ such that $|x_i\z-x_n|>\eps$ for all $i<n$.

This procedure might terminate on the $n$-th step if there is no point $x_n$ such that $|x_i\z-x_n|>\eps$ for all $i<n$.
In this case, the set $\{x_1,\ldots,x_{n-1}\}$ is an $\eps$-net in $\c{M}$ --- the lemma is proved.

If the procedure does not terminate, we get an infinite sequence of points $x_1,x_2,\ldots$ such that $|x_i-x_j|>\eps$ for all $i\ne j$.
Any of its subsequence has the same property; in particular none of its subsequences is Cauchy, so it cannot converge --- a contradiction.
\qeds

A topological space is called \index{separable space}\emph{separable} if it contains a countable dense subset.

\begin{thm}{Corollary}\label{cor:comp>separable}
Sequentially compact metric spaces are separable.
\end{thm}

\parit{Proof.}
Let $\c{M}$ be a sequentially compact metric space.

By \ref{lem:net}, for each positive integer $n$,
we can choose a finite $\tfrac1n$-net $N_n\subset\c{M}$.
It remains to observe that the union $N_1\cup N_2\cup \ldots$ is countable and dense.
\qeds

\section{Countable base}

\begin{thm}{Proposition}\label{cor:comp>2count}
Any sequentially compact metric space has a countable base.
\end{thm}

Topological spaces that admit a countable base are called \index{second-countable space}\emph{second-countable}.\footnote{A space is called \index{first-countable space}\emph{first-countable} if for each point $x$ there exists a sequence of neighbourhoods $N_1$, $N_2,\ldots$ such that for any open set $V\ni x$ we can choose $n$ such that $V\supset N_n\ni x$. For example, any metric space is first-countable; indeed, one can take $N_n=\Ball(x,\tfrac1n)$.}
So the proposition states that \textit{any sequentially compact metric space is second-countable}.

\parit{Proof.}
Let $\c{M}$ be a sequentially compact metric space.
By \ref{cor:comp>separable}, we can choose a countable dense subset $A\subset\c{M}$.

Consider the set of all balls $\Ball(a,\tfrac 1n)$ for $a\in A$ and positive integers~$n$.
Note that this set is countable; it remains to show that it forms a base in~$\c{M}$.

Let $x$ be a point in an open set $V$.
Then $\Ball(x,\eps)\subset V$ for some $\eps>0$.
Choose a positive integer $n>\tfrac 2\eps$.
Since $A$ is everywhere dense, we can choose $a\in A$ so that $|a-x|< \tfrac1n$.
By the triangle inequality, $x\in \Ball(a,\tfrac1n)\subset \Ball(x,\eps)$;
in particular,
\[x\in \Ball(a,\tfrac1n)\subset V.\]
It remains to apply \ref{prop:base-any}.
\qeds

\begin{thm}{Lemma}\label{lem:lind}
Let $\c{X}$ be a topological space with a countable base.
Then any open cover of $\c{X}$ has a countable subcover.
\end{thm}

A topological space is called \index{Lindelöf space}\emph{Lindelöf} if every open cover contains a countable subcover;
so the lemma says that \textit{any space with a countable base is Lindelöf}.

\parit{Proof.}
Choose an open cover $\{V_\alpha\}$.
Let $\{B_1,B_2,\ldots\}$ be a countable base of $\c{X}$.
By \ref{ex:base-nbhd}, for any $x\in\c{X}$ we can choose $i=i(x)$ such that $x\in B_i\subset V_\alpha$ for some $\alpha$.
Denote by $S$ the set of all indexes that appear as $i(x)$ for some $x$.
Then $\{B_i\}_{i\in S}$ is a countable open cover that is inscribed in $\{V_\alpha\}$.
That is, for every $B_i$ there is $\alpha_i$ such that $B_i\subset V_{\alpha_i}$.
It remains to observe that $\{V_{\alpha_i}\}_{i\in S}$ is a countable cover of $\c{X}$.
\qeds

\section[\texorpdfstring{Sequential compactness $\Rightarrow$  compactness}{Sequential compactness ⇒ compactness}]{Sequential compactness $\bm{\Rightarrow}$ compactness}\label{sec:seq-comp-if}

\parit{Proof of the if part in \ref{prop:seq-comp-metr}.}
Choose an open cover of $\c{M}$.
By \ref{lem:lind}, we can assume that it is countable; denote it by $\{V_1,V_2,\ldots\}$.
Assume it has no finite subcover.
Then we can choose a sequence of points $x_1,x_2,\ldots\in\c{M}$ such that
\[x_n\notin  V_1\cup \ldots\cup V_n\]
for any $n$.

Since $\c{M}$ is sequentially compact, a subsequence of $x_1,x_2,\ldots$ has a limit, say $x$,
and $x\in V_n$ for some $n$.
It follows that $x_i\in V_n$ for an infinite set of indices $i$, but by construction, $x_i\notin V_n$ for all $i\ge n$ --- a contradiction.
\qeds

\begin{thm}{Exercise}\label{ex:compact-net}
Show that a complete metric space $\c{M}$ is compact if for any $\eps>0$ there is a compact $\eps$-net in $\c{M}$.
\end{thm}

\begin{thm}{Advanced exercise}\label{ex:non-contracting-map}
Let $\c{M}$ be a compact metric space and
$f\:\c{M}\z\to \c{M}$
be a \index{distance-noncontracting map}\emph{distance-noncontracting map}; that is,
\[|f(x)-f(y)|_{\c{M}}\ge |x-y|_{\c{M}}\eqlbl{eq:distance-noncontracting}\]
for any $x,y\in\c{M}$.
Show that $f$ is an \index{isometry}\emph{isometry};
that is, $f$ is a bijection and equality always holds in \ref{eq:distance-noncontracting}.
\end{thm}

\section{An application}

\begin{thm}{Theorem}\label{thm:isoperim}
Among all plane curves of given length $L$, only circles bound the greatest area.
\end{thm}

Since a circle of length $L$ bounds area $\tfrac{L^2}{4\cdot \pi}$, the statement can be reformulated in a more algebraic manner: \textit{a plane curve $\gamma$ of given length $L$ bounds area  $a\le \tfrac{L^2}{4\cdot \pi}$, and, in the case of equality $\gamma$ is a circle.}
This statement is called {}\emph{isoperimetric inequality}.

Given a plane curve $\gamma$ denote by $F_\gamma$ its {}\emph{convex hull}; that is, minimal convex set that includes $\gamma$.
A figure bounded by $\gamma$ must lie in $F_\gamma$.
Furthermore, if $\gamma$ is a circle then it bounds $F_\gamma$.
Therefore, it is sufficient to prove the following two statements:

\begin{thm}{Lemma}\label{clm:isoperim-exist}
Among all closed plane curves of length $\le L$ there is a curve $\gamma$ that maximizes the area of $F_\gamma$.
\end{thm}

\begin{thm}{Lemma}\label{clm:isoperim-circle}
If a closed plane curve $\gamma$ of length $\le L$ maximizes the area of $F_\gamma$, then $\gamma$ must be a circle of length $L$.
\end{thm}

Note that \ref{clm:isoperim-circle} does not say that a circle maximizes the area, it only says that no other curve does.
But together with \ref{clm:isoperim-exist} it proves the theorem.

\parit{Proof of \ref{clm:isoperim-exist}.}
A closed plane curve $\gamma$ with length $\le L$ can be described by a parametrization $\gamma\:[0,L]\to\RR^2$ such that $\gamma(0)=\gamma(L)$ and
\[|\gamma(t_0)-\gamma(t_1)|_{\RR^2}\le |t_0-t_1|.\]
Without loss of generality, we can assume that $\gamma(0)=\gamma(L)=0\in\RR^2$.
Consider the space $\Gamma$ of all such parametrized curves with the \index{sup-norm metric}\emph{sup-norm metric} defined by
\[|\gamma_1-\gamma_2|_\Gamma= \max\set{|\gamma_1(t)-\gamma_2(t)|_{\RR^2}}{t\in[0,L]}
\eqlbl{eq:Gamma-metric}\]

\begin{thm}{Exercise}\label{ex:aa}
Prove the following statements:
\begin{subthm}{ex:aa:def}
The formula defines a metric on $\Gamma$; in particular, the maximum in the right-hand side in \ref{eq:Gamma-metric} is defined.
\end{subthm}

\begin{subthm}{ex:aa:complete}
The space $\Gamma$ is complete.
\end{subthm}

\begin{subthm}{ex:aa:compact}
The space $\Gamma$ is compact.
\end{subthm}

\end{thm}


It remains to prove the following sublemma and apply \ref{ex:fun(compact):min-max}.
\qeds

\begin{thm}{Sublemma}\label{clm:dido3}
The function $\Gamma\to\RR$ defined by
$\gamma\mapsto \area F_\gamma$ is continuous.
\end{thm}

\parit{Proof.}
It is sufficient to show that

for any $\gamma_1,\gamma_2\in \Gamma$.


Note that $F_{\gamma_2}$ lies in an $\tfrac Ln$-neighborhood of $F_{\gamma_1}$.
Moreover for any $p\in F_{\gamma_2}\setminus F_{\gamma_1}$ there is $x\in\partial F_{\gamma_1}$ such that $|p-x|<\tfrac Ln$.
Observe that length of $\partial F_{\gamma_1}$ cannot exceed~$L$.
Whence we can choose an $n$ points $x_1,\ldots,x_n$ such that for any $x\in \partial F_{\gamma_1}$ there is $x_i$ such that $|x-x_i|<\tfrac Ln$.
It follows that $|p-x_i|<2\cdot\tfrac Ln$ for some $i$, and
\[F_{\gamma_2}\setminus F_{\gamma_1}\subset \bigcup_{i=1}^{n}\Ball(x_i,2\cdot \tfrac Ln).\]
Since $\area\Ball(x,r)_{\RR^2}=\pi\cdot r^2$ we get
\[\area (F_{\gamma_2}\setminus F_{\gamma_1})\le  n\cdot\pi \cdot (2\cdot \tfrac Ln)^2,\]
and \ref{eq:area-cont} follows.
\qeds

}

We just described a very common type of application of compactness, which is worth knowing.
Note that we used a huge compact space $\Gamma$;
say, $\Gamma$ cannot be embedded into $\RR^n$ for any $n$.
In particular, one cannot apply standard calculus to $\Gamma$.
