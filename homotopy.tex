\chapter{Homotopy}



\section{Homotopies}

\begin{thm}{Definition}
Let $f,g\:\c{X}\to\c{Y}$ be continuous maps between topological spaces.
A continuous map $(x,t)\mapsto h_t(x)$ from $\c{X}\times[0,1]$ to $\c{Y}$ is called a \index{homotopy}\emph{homotopy} from $f$ to $g$ if
\[
h_0(x)=f(x)
\quad\text{and}\quad
h_1(x)=g(x)
\]
for all $x\in\c{X}$.

Suppose that in addition all the maps \emph{agree} on a subset $A\subset \c{X}$; that is,
$f(a)=g(a)=h_t(a)$ for any $a\in A$ and $t\in [0,1]$.
Then we say that $h_t$ is a homotopy from $f$ to $g$ \emph{relative} $A$.
\end{thm}

The map $(x,t)\mapsto h_t(x)$ should be considered as a one-parameter family of maps; we choose notation that helps to visualize this.

Two maps $f,g\:\c{X}\to\c{Y}$ are called \emph{homotopic} (\emph{homotopic relative} $A\subset \c{X}$) if there is a homotopy from $f$ to $g$ (respectively, homotopy relative $A$).
In these cases, we write $f\sim g$ and $f\sim g \rel A$, respectively.



\parit{Proof.}
\emph{Reflexive.} For $f\:\c{X}\to\c{Y}$ set $H(x,t)=f(x)$.
Then $H$ is a homotopy rel $A$ from $f$ to $f$.

\emph{Symmetric.} Let $H$ be a homotopy rel $A$ from $f_0$ to $f_1$.
Set $H'(x,t)=H(x,1-t)$.
Then $H'$ is a homotopy rel $A$ from $f_1$ to $f_0$.

\emph{Transitive.} Let $H$ be a homotopy rel $A$ from $f_0$ to $f_1$ and $G$ a homotopy rel $A$ from $f_1$ to $f_2$.
Define $F\:\c{X}\times[0,1]\to\c{Y}$ by
\[
F(x,t)=
\begin{cases}
H(x,2t)&\text{if } t\le \tfrac12,\\
G(x,2t-1)&\text{if } t\ge \tfrac12.
\end{cases}
\]
Since $H(x,1)=f_1(x)=G(x,0)$, the two formulas agree on $\c{X}\times\{\tfrac12\}$; hence $F$ is continuous (apply the standard pasting argument, cf.\ Exercise 4.3 in the notes).
Clearly $F$ is a homotopy from $f_0$ to $f_2$ and it is rel $A$ because both $H$ and $G$ are rel $A$.
\qeds

\section{Homotopy type}

\begin{thm}{Definition}
Two spaces $\c{X}$ and $\c{Y}$ are said to have the same \index{homotopy type}\emph{homotopy type} if there are continuous maps
\[
f\:\c{X}\to\c{Y}
\quad\text{and}\quad
g\:\c{Y}\to\c{X}
\]
such that
\[
g\circ f\sim \id_{\c{X}}
\quad\text{and}\quad
f\circ g\sim \id_{\c{Y}}.
\]
In this case $f$ is called a \index{homotopy equivalence}\emph{homotopy equivalence} and $g$ is called its \emph{homotopy inverse}.
\end{thm}

\begin{thm}{Definition}
A topological space $\c{X}$ is called \index{contractible space}\emph{contractible} if it has the same homotopy type as a one-point space.
Equivalently, $\c{X}$ is contractible if $\id_{\c{X}}$ is homotopic to a constant map $\c{X}\to\c{X}$.
\end{thm}

\section{Retracts and deformation retracts}

\begin{thm}{Definition}
Let $A\subset \c{X}$.
A continuous map $r\:\c{X}\to A$ is called a \index{retraction}\emph{retraction} if $r|_A=\id_A$.
In this case $A$ is called a \index{retract}\emph{retract} of $\c{X}$.
\end{thm}

\begin{thm}{Definition}
Let $A\subset \c{X}$ and denote by $\iota\:A\hookrightarrow \c{X}$ the inclusion map.

We say that $A$ is a \index{deformation retract}\emph{deformation retract} of $\c{X}$ if there is a retraction $r\:\c{X}\to A$ and a homotopy
\[
H\:\c{X}\times[0,1]\to \c{X}
\]
such that $H(x,0)=x$ for all $x\in\c{X}$ and $H(x,1)=\iota\circ r(x)$ for all $x\in\c{X}$.

If, in addition, $H(a,t)=a$ for all $a\in A$ and $t\in[0,1]$, then $A$ is called a \index{strong deformation retract}\emph{strong deformation retract} of $\c{X}$.
\end{thm}

\begin{thm}{Proposition}
If $A$ is a deformation retract of $\c{X}$, then $\c{X}$ and $A$ have the same homotopy type.
\end{thm}

\parit{Proof.}
Let $r\:\c{X}\to A$ be the retraction and $\iota\:A\hookrightarrow \c{X}$ the inclusion.
By definition, we have a homotopy $H$ from $\id_{\c{X}}$ to $\iota\circ r$, hence $\id_{\c{X}}\sim \iota\circ r$.
On the other hand, $r\circ \iota=\id_A$ since $r$ is a retraction.
Therefore $\iota$ and $r$ define a homotopy equivalence between $A$ and $\c{X}$.
\qeds
