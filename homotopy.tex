\chapter{Homotopy}

\section{Homotopy}\label{sec:homotopy}

Two continuous maps $f,g\:\c{X}\to \c{Y}$ are called \index{homotopic}\emph{homotopic} (briefly, $f\sim g$) if $f$ can be continuously deformed into $g$ within the class of continuous maps.
Formally, this means that there exists a continuous map
\[
H\:\c{X}\times[0,1]\to \c{Y}
\]
such that $H(x,0)=f(x)$ and $H(x,1)=g(x)$ for all $x\in \c{X}$.
The map $H\:\c{X}\times[0,1]\to \c{Y}$ is called \emph{homotopy} from $f$ to $g$.

It is often convinient to think of the homotopy $H$ is a one-parameter family of maps $h_t\:\c{X}\to \c{Y}$ defined by $h_t(x)=H(x,t)$.


\begin{thm}{Exercise}
Show that any two continous maps $\c{X}\to\RR^2$ are homotopic.
\end{thm}

\begin{thm}{Exercise}
Let $f_0, f_1\:\c{X}\to\c{Y}$ and $g_0,g_1\:\c{Y}\to \c{Z}$ be
four maps between topological spaces.
Assume that $f_0\sim f_1$ and $g_0\sim g_1$.
Show that $g_0\circ f_0\sim g_1\circ f_1$.
\end{thm}


\section{Operations on homotopies}\label{sec:homot-operations}

Observe that
\begin{itemize}
\item the \index{constant homotopy}\emph{constant homotopy} $H(x,t)=f(x)$ joins $f$ with itself;
\item if $H$ is a homotopy from $f$ with $g$, then the
\index{time-reversed}\emph{time-reversed} homotopy defined by
\[
\bar H(x,t)=H(x,1-t)
\]
is a homotopy from $g$ to $f$;
\item Consider two homotopies $H$ and $G$ from $f$ to $g$ and from $g$ to $h$, respectively.
Then their \index{concatenation of homotopies}\emph{concatenation} $F=H*G$ defined by
\[
F(x,t)=
\begin{cases}
H(x,2\cdot t),& t\le \tfrac12,\\
G(x,2\cdot t-1),& t\ge \tfrac12,
\end{cases}
\]
defines a homotopy from $f$ with $h$.
(By \ref{ex:u-continuous:closed}, $F$ is continuous.)
\end{itemize}

\section{Homotopy classes}

The previous section implies that $\sim$ is an equivalence relation on the set of continuous maps $\c{X}\to\c{Y}$; in other words, the following properties hold for any continuous maps $f,g,h\: \c{X}\to\c{Y}$:
\begin{itemize}
 \item $f\sim f$;
 \item if $f\sim g$, then $g\sim f$;
 \item if $f\sim g$, and $g\sim h$, then $f\sim h$.
\end{itemize}

The equivalence classes for $\sim$ are called \index{homotopy class}\emph{homotopy classes};
the equivalence class of a map $f\:\c{X}\to\c{Y}$ will be denoted by $[f]$.
Given two spaces $\c{X}$ and $\c{Y}$, the set of all homotopy classes will be denoted by
$\pi(\c{X},\c{Y})$.

Often the set $\pi(\c{X},\c{Y})$ admits a direct description; this is precisely its advantage over the vast set of all continuous maps $\c{X}\to \c{Y}$.
Note that any homeomorphism $\c{Y}\to \c{Y}'$ defines a bijection between homotopy classes $\pi(\c{X},\c{Y})\to \pi(\c{X},\c{Y}')$ for any given space $\c{X}$.
So, stydying the homotopy classes may help to distiguish topological spaces.


This construction is meaningful even for very simple choices of $\c{X}$.
If $\c{X}$ is the one-point space, then $\pi(\c{X},\c{Y})$ is also denoted by $\pi_0(\c{Y})$.
This is the main subject of this chapter.
If $\pi(\SS^1,\c{Y})$ is closely related to the fundamental group $\pi_1(\c{Y})$, which is the main subject of the following chapters.

The idea to use homotopy classes to distiguish topological spaces will guide us in the right direction.
However, the true motivation to consider homotopy classes is different; it will described in Capter \ref{chap:algebra}.

\section{Homotopy type}

\begin{thm}{Definition}
Two spaces $\c{X}$ and $\c{Y}$ are said to have the same \index{homotopy type}\emph{homotopy type} if there are continuous maps
\[
f\:\c{X}\to\c{Y}
\quad\text{and}\quad
g\:\c{Y}\to\c{X}
\]
such that
\[
g\circ f\sim \id_{\c{X}}
\quad\text{and}\quad
f\circ g\sim \id_{\c{Y}}.
\]
In this case $f$ is called a \index{homotopy equivalence}\emph{homotopy equivalence} and $g$ is called its \emph{homotopy inverse}.
\end{thm}

\begin{thm}{Definition}
A topological space $\c{X}$ is called \index{contractible space}\emph{contractible} if it has the same homotopy type as a one-point space.
Equivalently, $\c{X}$ is contractible if $\id_{\c{X}}$ is homotopic to a constant map $\c{X}\to\c{X}$.
\end{thm}

\section{Retracts and deformation retracts}

\begin{thm}{Definition}
Let $A\subset \c{X}$.
A continuous map $r\:\c{X}\to A$ is called a \index{retraction}\emph{retraction} if $r|_A=\id_A$.
In this case $A$ is called a \index{retract}\emph{retract} of $\c{X}$.
\end{thm}

\begin{thm}{Definition}
A set $A$ in a topological space $\c{X}$ is called \index{deformation retract}\emph{deformation retract} of $\c{X}$ if there is a retraction $r\:\c{X}\to A$ and a homotopy
\[
H\:\c{X}\times[0,1]\to \c{X}
\]
from the identity map $\id_{\c{X}}$ to $r$.

If, in addition, $H(a,t)=a$ for all $a\in A$ and $t\in[0,1]$, then $A$ is called a \index{strong deformation retract}\emph{strong deformation retract} of $\c{X}$.
\end{thm}

\begin{thm}{Proposition}
If $A$ is a deformation retract of $\c{X}$, then $\c{X}$ and $A$ have the same homotopy type.
\end{thm}

\parit{Proof.}
Let $r\:\c{X}\to A$ be the retraction and $\iota\:A\hookrightarrow \c{X}$ the inclusion.
By definition, we have a homotopy $H$ from $\id_{\c{X}}$ to $\iota\circ r$, hence $\id_{\c{X}}\sim \iota\circ r$.
On the other hand, $r\circ \iota=\id_A$ since $r$ is a retraction.
Therefore $\iota$ and $r$ define a homotopy equivalence between $A$ and $\c{X}$.
\qeds

\section{Contractible spaces}

A topological space $\c{X}$ is called \index{contactable space}\emph{contactable} if the identity map $\id_\c{X}\:\c{X}\z\to \c{X}$ is homotopic to a constant map;
that is, there is a homotopy $h_t\:\c{X}\to\c{X}$ such that $h_0(x)=x$ and $h_1(x)=p$ for some fixed point $p\in\c{X}$ and any $t$.

\begin{thm}{Exercise}
Show that any convex subset of the Euclidean space is contractible.
\end{thm}


\begin{thm}{Exercise}
Show that any contractible space is path-connected.
\end{thm}

Later we will show that the circle $\mathbb{S}^1$ is an example of path-connected space that is not contractible.

\begin{subthm}{}
Show that any two continuous maps from a topological space to $\c{X}$ are homotopic.
\end{subthm}

\begin{subthm}{}
Show that any two continuous maps from $\c{X}$ to a path-connected space are homotopic.
\end{subthm}

\end{thm}

\section{Homotopy type}

Two topological spaces $\c{X}$ and $\c{Y}$ have the same \index{homotopy type}\emph{homotopy type} (briefly $\c{X}\sim\c{Y}$) if there are continuous maps
$f\: \c{X}\to \c{Y}$ and $h\:\c{Y}\to \c{X}$ such that $h\circ f\sim \id_\c{X}$ and $f\circ h\sim \id_\c{Y}$.

\begin{thm}{Exercise}
Show that a topological space $\c{X}$ is contractible if and only if it has the same homotopy type with the one-point space.
\end{thm}
